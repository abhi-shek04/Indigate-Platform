    84→/**
    85→ * Returns a 0–100 percentage of how many resume sections have at least some
    86→ * data filled. Pure function — used to render the sidebar progress bar.
    87→ */
    88→function computeProgress(data: ResumeData): number {
    89→  const filled: boolean[] = [
    90→    !!(data.name?.trim() || data.email?.trim() || data.phone?.trim()),
    91→    data.education.length > 0,
    92→    data.activities.length > 0,
    93→    data.awards.length > 0,
    94→    data.projects.length > 0,
    95→    data.skills.length > 0,
    96→    (data.skillsExcelSummary?.length ?? 0) > 0,
    97→    !!(data.currentJlpt || data.expectedJlpt || data.otherLanguages?.trim()),
    98→    !!(data.japanMotivation?.whyJapan?.trim() ||
    99→      data.japanMotivation?.careerInJapan?.trim() ||
   100→      data.japanMotivation?.challenges?.trim()),
   101→    !!(data.selfPr?.trim() || data.hobbies?.trim()),
   102→  ];
   103→  return Math.round(
   104→    (filled.filter(Boolean).length / filled.length) * 100,
   105→  );
   106→}
