   180→function computeCompletion(c: CandidateProfileDTO | null): number {
   181→  if (!c) return 0;
   182→  const checks: boolean[] = [
   183→    !!c.fullName?.trim(),
   184→    !!c.phone?.trim(),
   185→    !!c.bio?.trim(),
   186→    !!c.location?.trim(),
   187→    c.jlptLevel !== "NONE",
   188→    (c.skills?.length ?? 0) >= 3,
   189→    !!c.resumeUrl,
   190→    c.experienceYears > 0,
   191→    !!c.education && c.education.length > 0,
   192→    !!c.linkedinUrl?.trim(),
   193→  ];
   194→  const filled = checks.filter(Boolean).length;
   195→  return Math.round((filled / checks.length) * 100);
   196→}

   407→  const refreshAuth = useApp((s) => s.refreshAuth);
   408→  const [apps, setApps] = useState<ApplicationDTO[] | null>(null);
   409→  const [loading, setLoading] = useState(true);
   410→  const [busyId, setBusyId] = useState<string | null>(null);
   411→
   412→  const load = useCallback(async () => {
   413→    setLoading(true);
   414→    try {
   415→      const res = await api<{ applications: ApplicationDTO[] }>(
   416→        "/api/applications",
   417→      );
   418→      setApps(res.applications);
   419→    } catch {
   420→      setApps([]);
   421→    } finally {
   422→      setLoading(false);
   423→    }
   424→  }, []);
   425→
   426→  useEffect(() => {
   427→    load();
   428→  }, [load]);
   429→
   430→  async function withdraw(a: ApplicationDTO) {
   431→    setBusyId(a.id);
   432→    try {
   433→      await api(`/api/applications/${a.id}/status`, {
   434→        method: "PATCH",
   435→        body: JSON.stringify({ status: "WITHDRAWN" }),
   436→      });
   437→      toast.success("Application withdrawn.");
   438→      await load();
   439→      await refreshAuth();
   440→    } catch (e) {
   441→      toast.error(e instanceof Error ? e.message : "Failed to withdraw.");
   442→    } finally {
   443→      setBusyId(null);
   444→    }
   445→  }
   446→
   447→  if (loading) {
   448→    return (
   449→      <div className="space-y-4">
   450→        <CardSkeleton lines={6} />
   451→      </div>
   452→    );
   453→  }
   454→
   455→  if (!apps || apps.length === 0) {
   456→    return (
   457→      <EmptyState
   458→        icon={FileText}
   459→        title={t("dash.apps.empty")}
   460→        description="Browse curated roles in Japan and apply with one click."
   461→        action={
   462→          <Button
   463→            className="bg-brand-gradient text-white hover:opacity-90"
   464→            onClick={() => navigate("jobs")}
   465→          >
   466→            <Briefcase className="h-4 w-4" />
   467→            {t("dash.apps.apply")}
   468→          </Button>
   469→        }
   470→      />
   471→    );
   472→  }
   473→
   474→  return (
   475→    <div className="space-y-4">
   476→      <SectionCard
   477→        title={`${apps.length} application${apps.length === 1 ? "" : "s"}`}
   478→        bodyClassName="p-0"
   479→      >
   480→        <div className="max-h-[70vh] overflow-y-auto scroll-area">
   481→          <Table>
   482→            <TableHeader className="sticky top-0 bg-card z-10">
   483→              <TableRow>
   484→                <TableHead className="pl-5 sm:pl-6">Role</TableHead>
   485→                <TableHead className="hidden sm:table-cell">Applied</TableHead>
   486→                <TableHead>Status</TableHead>
   487→                <TableHead className="text-right pr-5 sm:pr-6">Action</TableHead>
   488→              </TableRow>
   489→            </TableHeader>
   490→            <TableBody>
   491→              {apps.map((a) => (
   492→                <Fragment key={a.id}>
   493→                <TableRow>
   494→                  <TableCell className="pl-5 sm:pl-6">
   495→                    <button
   496→                      onClick={() =>
   497→                        a.job && navigate("job-detail", { jobId: a.job.id })
   498→                      }
   499→                      className="text-left"
   500→                    >
   501→                      <p className="font-semibold text-sm hover:text-crimson transition-colors">
   502→                        {a.job?.title ?? "Job removed"}
   503→                      </p>
   504→                      <p className="text-xs text-muted-foreground">
   505→                        {a.job?.company?.companyName} ·{" "}
   506→                        {a.job?.location}
   507→                      </p>
   508→                    </button>

   511→                    {formatDate(a.appliedAt, locale)}
   512→                  </TableCell>
   513→                  <TableCell>
   514→                    <Badge
   515→                      variant="outline"
   516→                      className={cn("font-semibold", STATUS_BADGE[a.status])}
   517→                    >
   518→                      {t(`status.${a.status}`)}
   519→                    </Badge>
   520→                  </TableCell>
   521→                  <TableCell className="text-right pr-5 sm:pr-6">
   522→                    {a.status !== "WITHDRAWN" && (
   523→                      <AlertDialog>
   524→                        <AlertDialogTrigger asChild>
   525→                          <Button
   526→                            variant="ghost"
   527→                            size="sm"

   537→                              Withdraw this application?
   538→                            </AlertDialogTitle>

   540→                              You're about to withdraw your application for{" "}
   541→                              <span className="font-medium text-foreground">
   542→                                {a.job?.title}
   543→                              </span>{" "}
   544→                              at{" "}
   545→                              <span className="font-medium text-foreground">
   546→                                {a.job?.company?.companyName}

   551→                          <AlertDialogFooter>

   553→                              {t("common.cancel")}

   556→                              className="bg-destructive text-white hover:bg-destructive/90"
   557→                              onClick={() => withdraw(a)}

   559→                              Withdraw
   560→                            </AlertDialogAction>
   561→                          </AlertDialogFooter>
   562→                        </AlertDialogContent>
   563→                      </AlertDialog>
   564→                    )}
   565→                  </TableCell>
   566→                </TableRow>
   567→                {a.status === "INTERVIEWED" && a.interviewDate && (
   568→                  <TableRow className="bg-violet-50/50 dark:bg-violet-950/20">
   569→                    <TableCell colSpan={4} className="px-5 sm:px-6 py-3">
   570→                      <InterviewInfo app={a} />
   571→                    </TableCell>
   572→                  </TableRow>
   573→                )}
   574→                </Fragment>
   575→              ))}
   576→            </TableBody>
   577→          </Table>
   578→        </div>
   579→      </SectionCard>
   580→    </div>
   581→  );
   582→}

   586→function Profile() {
   587→  const candidate = useApp((s) => s.candidate);
   588→  const refreshAuth = useApp((s) => s.refreshAuth);
   589→  const { t } = useT();
   590→
   591→  const [form, setForm] = useState({
   592→    fullName: candidate?.fullName ?? "",
   593→    phone: candidate?.phone ?? "",
   594→    location: candidate?.location ?? "",
   595→    bio: candidate?.bio ?? "",
   596→    linkedinUrl: candidate?.linkedinUrl ?? "",
   597→    experienceYears: candidate?.experienceYears ?? 0,
   598→    jlptLevel: (candidate?.jlptLevel ?? "NONE") as JLPTLevel,
   599→    skills: candidate?.skills ?? [],
   600→    education: (candidate?.education ?? []) as EducationEntry[],
   601→  });
   602→  const [saving, setSaving] = useState(false);
   603→  const [openToWork, setOpenToWork] = useState<boolean>(
   604→    candidate?.openToWork ?? true,
   605→  );
   606→  const [toggling, setToggling] = useState(false);
   607→
   608→  // re-sync when store candidate changes (e.g. after refreshAuth)
   609→  useEffect(() => {
   610→    if (!candidate) return;
   611→    setForm((prev) => ({
   612→      ...prev,
   613→      fullName: candidate.fullName,
   614→      phone: candidate.phone ?? "",
   615→      location: candidate.location ?? "",
   616→      bio: candidate.bio ?? "",
   617→      linkedinUrl: candidate.linkedinUrl ?? "",
   618→      experienceYears: candidate.experienceYears,
   619→      jlptLevel: candidate.jlptLevel,
   620→      skills: candidate.skills,
   621→      education: candidate.education ?? [],
   622→    }));
   623→    setOpenToWork(candidate.openToWork);
   624→  }, [candidate?.updatedAt]);
