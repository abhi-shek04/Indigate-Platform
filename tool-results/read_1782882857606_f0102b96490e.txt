    58→    58→function StatCard({
    59→    59→  value,
    60→    60→  suffix,
    61→    61→  label,
    62→    62→  delay,
    63→    63→}: {
    64→    64→  value: number;
    65→    65→  suffix?: string;
    66→    66→  label: string;
    67→    67→  delay: number;
    68→    68→}) {
    69→    69→  const count = useCountUp(value, 1800);
    70→    70→  const ref = useRef<HTMLDivElement>(null);
    71→    71→  const [visible, setVisible] = useState(false);
    72→    72→
    73→    73→  useEffect(() => {
    74→    74→    const el = ref.current;
    75→    75→    if (!el) return;
    76→    76→    const obs = new IntersectionObserver(
    77→    77→      ([e]) => {
    78→    78→        if (e.isIntersecting) {
    79→    79→          setVisible(true);
    80→    80→          obs.disconnect();
    81→    81→        }
    82→    82→      },
    83→    83→      { threshold: 0.3 },
    84→    84→    );
    85→    85→    obs.observe(el);
    86→    86→    return () => obs.disconnect();
    87→    87→  }, []);
    88→    88→
    89→    89→  return (
    90→    90→    <motion.div
    91→    91→      ref={ref}
    92→    92→      initial={{ opacity: 0, y: 20 }}
    93→    93→      animate={visible ? { opacity: 1, y: 0 } : {}}
    94→    94→      transition={{ duration: 0.6, ease: easeOutExpo, delay }}
    95→    95→      className="text-center"
    96→    96→    >
    97→    97→      <div className="font-display text-4xl sm:text-5xl font-extrabold tracking-tight">
    98→    98→        <span className="text-gradient-brand">
    99→    99→          {count.toLocaleString()}
   100→   100→          {suffix}
   101→   101→        </span>
   102→   102→      </div>
   103→   103→      <p className="mt-1.5 text-sm font-medium text-muted-foreground">{label}</p>
   104→   104→    </motion.div>
   105→   105→  );
   106→   106→}
