   167→function computeCompletion(c: CandidateProfileDTO | null): number {
   168→  if (!c) return 0;
   169→  const checks: boolean[] = [
   170→    !!c.fullName?.trim(),
   171→    !!c.phone?.trim(),
   172→    !!c.bio?.trim(),
   173→    !!c.location?.trim(),
   174→    c.jlptLevel !== "NONE",
   175→    (c.skills?.length ?? 0) >= 3,
   176→    !!c.resumeUrl,
   177→    c.experienceYears > 0,
   178→    !!c.education && c.education.length > 0,
   179→    !!c.linkedinUrl?.trim(),
   180→  ];
   181→  const filled = checks.filter(Boolean).length;
   182→  return Math.round((filled / checks.length) * 100);
   183→}

   369→  const load = useCallback(async () => {
   370→    setLoading(true);

   372→      const res = await api<{ applications: ApplicationDTO[] }>(
   373→        "/api/applications",

   375→      setApps(res.applications);

   377→      setApps([]);

   384→    load();

   387→  async function withdraw(a: ApplicationDTO) {

   391→        method: "PATCH",

   399→    } finally {
   400→      setBusyId(null);
   401→    }
   402→  }
   403→

   405→    return (
   406→      <div className="space-y-4">
   407→        <CardSkeleton lines={6} />

   415→        icon={FileText}
   416→        title={t("dash.apps.empty")}

   578→  function set<K extends keyof typeof form>(k: K, v: (typeof form)[K]) {

   582→  function addEducation() {
   583→    set("education", [

   837→function Resume() {

   840→  const { t, locale } = useT();

   844→  async function uploadFile(file: File) {
   845→    if (file.type !== "application/pdf" && !file.name.endsWith(".pdf")) {
   846→      toast.error("Resume must be a PDF.");
   847→      return;
   848→    }
   849→    if (file.size > 5 * 1024 * 1024) {
   850→      toast.error("Max file size is 5MB.");
   851→      return;
   852→    }
   853→    setBusy(true);
   854→    setProgress(0);
   855→    try {
   856→      const fd = new FormData();
   857→      fd.append("file", file);
   858→      fd.append("kind", "resume");
   859→      // simulate upload progress (fetch doesn't give us easy progress)
   860→      const fakeTimer = setInterval(() => {
   861→        setProgress((p) => Math.min(90, (p ?? 0) + 10));
   862→      }, 80);
   863→      const res = await fetch("/api/upload", { method: "POST", body: fd });
   864→      clearInterval(fakeTimer);
   865→      setProgress(100);
   866→      if (!res.ok) {
   867→        const data = await res.json().catch(() => null);
   868→        throw new Error(data?.error || "Upload failed.");
   869→      }
   870→      toast.success("Resume uploaded.");
   871→      await refreshAuth();
   872→    } catch (e) {
   873→      toast.error(e instanceof Error ? e.message : "Upload failed.");
   874→    } finally {
   875→      setBusy(false);
   876→      setTimeout(() => setProgress(null), 600);
   877→    }
   878→  }
   879→
   880→  async function removeResume() {
   881→    setBusy(true);
   882→    try {
   883→      // The candidate PUT schema accepts `resumeUrl: null`. (The upload route
   884→      // sets resumeName separately — we only null the URL here, which is what
   885→      // applicants see.)
   886→      await api("/api/candidates/me", {
   887→        method: "PUT",
   888→        body: JSON.stringify({ resumeUrl: null }),
   889→      });
   890→      toast.success("Resume removed.");
   891→      await refreshAuth();
   892→    } catch (e) {
   893→      toast.error(e instanceof Error ? e.message : "Failed to remove.");
   894→    } finally {
   895→      setBusy(false);
   896→    }
   897→  }
   898→
   899→  return (

   917→                <a
   918→                  href={candidate.resumeUrl}
   919→                  target="_blank"

   969→      <SectionCard title={t("dash.resume.upload")}>
