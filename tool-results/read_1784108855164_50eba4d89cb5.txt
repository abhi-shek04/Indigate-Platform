    87→/**
    88→ * Returns a 0–100 percentage of how many resume sections have at least some
    89→ * data filled. Pure function — used to render the sidebar progress bar.
    90→ */
    91→function computeProgress(data: ResumeData): number {
    92→  const filled: boolean[] = [
    93→    !!(data.name?.trim() || data.email?.trim() || data.phone?.trim()),
    94→    data.education.length > 0,
    95→    data.activities.length > 0,
    96→    data.awards.length > 0,
    97→    data.projects.length > 0,
    98→    data.skills.length > 0,
    99→    (data.skillsExcelSummary?.length ?? 0) > 0,
   100→    !!(data.currentJlpt || data.expectedJlpt || data.otherLanguages?.trim()),
   101→    !!(data.japanMotivation?.whyJapan?.trim() ||
   102→      data.japanMotivation?.careerInJapan?.trim() ||
   103→      data.japanMotivation?.challenges?.trim()),
   104→    !!(data.selfPr?.trim() || data.hobbies?.trim()),
   105→  ];
   106→  return Math.round(
   107→    (filled.filter(Boolean).length / filled.length) * 100,
   108→  );
   109→}
