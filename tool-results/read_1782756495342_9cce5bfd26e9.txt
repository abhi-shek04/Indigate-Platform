     1→"use client";
     2→
     3→import { useEffect, useState, useCallback } from "react";
     4→import { useApp } from "@/lib/store";
     5→import { useT } from "@/lib/use-t";
     6→import { api } from "@/lib/api-client";
     7→import { Button } from "@/components/ui/button";
     8→import { Input } from "@/components/ui/input";
     9→import { Label } from "@/components/ui/label";
    10→import { Textarea } from "@/components/ui/textarea";
    11→import { Badge } from "@/components/ui/badge";
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from "@/components/ui/select";
    19→import {
    20→  ArrowLeft,
    21→  Plus,
    22→  Trash2,
    23→  Save,
    24→  Printer,
    25→  FileText,
    26→  User,
    27→  GraduationCap,
    28→  Briefcase,
    29→  Award,
    30→  Heart,
    31→  Globe,
    32→  Sparkles,
    33→  Loader2,
    34→  X,
    35→  Download,
    36→} from "lucide-react";
    37→import { motion } from "framer-motion";
    38→import { toast } from "sonner";
    39→import { cn } from "@/lib/utils";
    40→import { PDFDownloadLink } from "@react-pdf/renderer";
    41→import { EnglishResumePDF } from "@/lib/pdf-templates/english-resume-pdf";
    42→import { JapaneseResumePDF } from "@/lib/pdf-templates/japanese-resume-pdf";
    43→import {
    44→  EMPTY_RESUME,
    45→  GENDER_OPTIONS,
    46→  NATIONALITY_OPTIONS,
    47→  INDIAN_STATES,
    48→  LANGUAGE_OPTIONS,
    49→  STATE_JA,
    50→  type ResumeData,
    51→  type ResumeEducation,
    52→  type ResumeProject,
    53→  type ResumeActivity,
    54→  type ResumeAward,
    55→} from "@/lib/resume-types";
    56→import { ResumePreview } from "./resume-preview";
    57→
    58→type Tab = "edit" | "preview-ja" | "preview-en";
    59→
    60→export function ResumeBuilder() {
    61→  const { t } = useT();
    62→  const navigate = useApp((s) => s.navigate);
    63→  const candidate = useApp((s) => s.candidate);
    64→  const user = useApp((s) => s.user);
    65→  const [data, setData] = useState<ResumeData>(EMPTY_RESUME);
    66→  const [loading, setLoading] = useState(true);
    67→  const [saving, setSaving] = useState(false);
    68→  const [tab, setTab] = useState<Tab>("edit");
    69→  const [langInput, setLangInput] = useState("");
    70→
    71→  const load = useCallback(async () => {
    72→    setLoading(true);
    73→    try {
    74→      const res = await api<{ resumeData: ResumeData | null }>(
    75→        "/api/candidates/me/resume",
    76→      );
    77→      if (res.resumeData) {
    78→        setData({ ...EMPTY_RESUME, ...res.resumeData });
    79→      } else if (candidate) {
    80→        setData({
    81→          ...EMPTY_RESUME,
    82→          name: candidate.fullName,
    83→          email: user?.email ?? "",
    84→          phone: candidate.phone ?? "",
    85→          nationality: "India",
    86→          languages: ["English"],
    87→          languagesJa: ["英語"],
    88→        });
    89→      }
    90→    } catch {
    91→      toast.error("Failed to load resume data.");
    92→    } finally {
    93→      setLoading(false);
    94→    }
    95→  }, [candidate?.id, user?.email]);
    96→
    97→  useEffect(() => {
    98→    load();
    99→  }, [load]);
   100→
   101→  async function save() {
   102→    setSaving(true);
   103→    try {
   104→      await api("/api/candidates/me/resume", {
   105→        method: "PUT",
   106→        body: JSON.stringify(data),
   107→      });
   108→      toast.success("Resume saved!");
   109→    } catch (e) {
   110→      toast.error(e instanceof Error ? e.message : "Failed to save.");
   111→    } finally {
   112→      setSaving(false);
   113→    }
   114→  }
   115→
   116→  function print() {
   117→    window.print();
   118→  }
   119→
   120→  function update<K extends keyof ResumeData>(key: K, value: ResumeData[K]) {
   121→    setData((prev) => ({ ...prev, [key]: value }));
   122→  }
   123→
   124→  // Language tag management
   125→  function addLanguage(lang: string) {
   126→    if (!lang.trim()) return;
   127→    const opt = LANGUAGE_OPTIONS.find((l) => l.value === lang);
   128→    if (data.languages.includes(lang)) return;
   129→    update("languages", [...data.languages, lang]);
   130→    update("languagesJa", [...data.languagesJa, opt?.labelJa ?? lang]);
   131→    setLangInput("");
   132→  }
   133→  function removeLanguage(idx: number) {
   134→    update(
   135→      "languages",
   136→      data.languages.filter((_, i) => i !== idx),
   137→    );
   138→    update(
   139→      "languagesJa",
   140→      data.languagesJa.filter((_, i) => i !== idx),
   141→    );
   142→  }
   143→
   144→  // Education
   145→  function addEducation() {
   146→    update("education", [
   147→      ...data.education,
   148→      { year: "", degree: "", field: "", institution: "" },
   149→    ]);
   150→  }
   151→  function updateEducation(i: number, patch: Partial<ResumeEducation>) {
   152→    update("education", data.education.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   153→  }
   154→  function removeEducation(i: number) {
   155→    update("education", data.education.filter((_, j) => j !== i));
   156→  }
   157→
   158→  // Projects
   159→  function addProject() {
   160→    update("projects", [
   161→      ...data.projects,
   162→      { period: "", name: "", description: "", techStack: "" },
   163→    ]);
   164→  }
   165→  function updateProject(i: number, patch: Partial<ResumeProject>) {
   166→    update("projects", data.projects.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   167→  }
   168→  function removeProject(i: number) {
   169→    update("projects", data.projects.filter((_, j) => j !== i));
   170→  }
   171→
   172→  // Activities
   173→  function addActivity() {
   174→    update("activities", [
   175→      ...data.activities,
   176→      { period: "", organization: "", role: "", duties: "" },
   177→    ]);
   178→  }
   179→  function updateActivity(i: number, patch: Partial<ResumeActivity>) {
   180→    update("activities", data.activities.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   181→  }
   182→  function removeActivity(i: number) {
   183→    update("activities", data.activities.filter((_, j) => j !== i));
   184→  }
   185→
   186→  // Awards
   187→  function addAward() {
   188→    update("awards", [
   189→      ...data.awards,
   190→      { year: "", title: "", description: "", organization: "" },
   191→    ]);
   192→  }
   193→  function updateAward(i: number, patch: Partial<ResumeAward>) {
   194→    update("awards", data.awards.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   195→  }
   196→  function removeAward(i: number) {
   197→    update("awards", data.awards.filter((_, j) => j !== i));
   198→  }
   199→
   200→  if (loading) {
   201→    return (
   202→      <div className="flex-1 grid place-items-center py-20">
   203→        <Loader2 className="h-8 w-8 animate-spin text-saffron" />
   204→      </div>
   205→    );
   206→  }
   207→
   208→  return (
   209→    <div className="mx-auto max-w-4xl px-4 sm:px-6 lg:px-8 py-8 print:max-w-none print:p-0">
   210→      {/* Header */}
   211→      <div className="print:hidden mb-6">
   212→        <button
   213→          onClick={() => navigate("candidate")}
   214→          className="inline-flex items-center gap-1.5 text-[13px] font-medium text-muted-foreground hover:text-foreground mb-3"
   215→        >
   216→          <ArrowLeft className="h-4 w-4" />
   217→          {t("common.back")}
   218→        </button>
   219→        <div className="flex items-center justify-between flex-wrap gap-4">
   220→          <div>
   221→            <h1 className="font-display text-2xl font-extrabold tracking-tight">
   222→              Resume Builder
   223→            </h1>
   224→            <p className="mt-1 text-[13px] text-muted-foreground">
   225→              Create a professional resume in English and Japanese 履歴書 format.
   226→            </p>
   227→          </div>
   228→          <div className="flex items-center gap-2 flex-wrap">
   229→            <Button onClick={save} disabled={saving} className="bg-brand-gradient text-white hover:brightness-110 font-bold text-[14px] h-10 px-5 border-0">
   230→              {saving ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Save className="mr-2 h-4 w-4" />}
   231→              Save
   232→            </Button>
   233→            <PDFDownloadLink
   234→              document={<EnglishResumePDF data={data} />}
   235→              fileName={`${data.name || "resume"}_EN.pdf`}
   236→            >
   237→              {({ loading }) => (
   238→                <Button variant="outline" size="sm" disabled={loading} className="font-bold text-[13px] h-10 px-4 border-saffron/30 text-crimson hover:bg-saffron/5">
   239→                  <Download className="h-4 w-4 mr-1.5" />
   240→                  {loading ? "…" : "EN PDF"}
   241→                </Button>
   242→              )}
   243→            </PDFDownloadLink>
   244→            <PDFDownloadLink
   245→              document={<JapaneseResumePDF data={data} />}
   246→              fileName={`${data.nameJa || data.name || "resume"}_JP.pdf`}
   247→            >
   248→              {({ loading }) => (
   249→                <Button variant="outline" size="sm" disabled={loading} className="font-bold text-[13px] h-10 px-4 border-saffron/30 text-crimson hover:bg-saffron/5">
   250→                  <Download className="h-4 w-4 mr-1.5" />
   251→                  {loading ? "…" : "JP PDF"}
   252→                </Button>
   253→              )}
   254→            </PDFDownloadLink>
   255→          </div>
   256→        </div>
   257→      </div>
   258→
   259→      {/* Tabs */}
   260→      <div className="print:hidden flex items-center gap-1 mb-6 p-1 rounded-xl bg-muted w-fit">
   261→        {[
   262→          { key: "edit" as const, label: "Edit", icon: User },
   263→          { key: "preview-ja" as const, label: "日本語 履歴書", icon: Globe },
   264→          { key: "preview-en" as const, label: "English", icon: FileText },
   265→        ].map((tb) => (
   266→          <button
   267→            key={tb.key}
   268→            onClick={() => setTab(tb.key)}
   269→            className={cn(
   270→              "inline-flex items-center gap-1.5 px-4 py-2 rounded-lg text-[13px] font-semibold transition-all",
   271→              tab === tb.key
   272→                ? "bg-card text-foreground shadow-sm"
   273→                : "text-muted-foreground hover:text-foreground",
   274→            )}
   275→          >
   276→            <tb.icon className="h-4 w-4" />
   277→            {tb.label}
   278→          </button>
   279→        ))}
   280→      </div>
   281→
   282→      {/* Edit form */}
   283→      {tab === "edit" && (
   284→        <motion.div
   285→          initial={{ opacity: 0, y: 8 }}
   286→          animate={{ opacity: 1, y: 0 }}
   287→          className="print:hidden space-y-6"
   288→        >
   289→          {/* Personal Info */}
   290→          <Section icon={User} title="Personal Information" desc="Your basic details — selectable fields make it faster">
   291→            <div className="grid sm:grid-cols-2 gap-4">
   292→              <Field label="Full name (English)">
   293→                <Input value={data.name} onChange={(e) => update("name", e.target.value)} placeholder="Abhishek" />
   294→              </Field>
   295→              <Field label="氏名 (Katakana reading)">
   296→                <Input value={data.nameJa ?? ""} onChange={(e) => update("nameJa", e.target.value)} placeholder="アビシェーク" />
   297→              </Field>
   298→              <Field label="Date of birth">
   299→                <Input type="date" value={data.dob ?? ""} onChange={(e) => update("dob", e.target.value)} />
   300→              </Field>
   301→              <Field label="Gender">
   302→                <Select value={data.gender ?? ""} onValueChange={(v) => update("gender", v)}>
   303→                  <SelectTrigger className="h-10">
   304→                    <SelectValue placeholder="Select gender" />
   305→                  </SelectTrigger>
   306→                  <SelectContent>
   307→                    {GENDER_OPTIONS.map((g) => (
   308→                      <SelectItem key={g.value} value={g.value}>
   309→                        {g.labelEn} ({g.labelJa})
   310→                      </SelectItem>
   311→                    ))}
   312→                  </SelectContent>
   313→                </Select>
   314→              </Field>
   315→              <Field label="Email">
   316→                <Input type="email" value={data.email} onChange={(e) => update("email", e.target.value)} placeholder="you@example.com" />
   317→              </Field>
   318→              <Field label="Phone">
   319→                <Input value={data.phone ?? ""} onChange={(e) => update("phone", e.target.value)} placeholder="+91 98765 43210" />
   320→              </Field>
   321→              <Field label="Nationality">
   322→                <Select value={data.nationality ?? ""} onValueChange={(v) => update("nationality", v)}>
   323→                  <SelectTrigger className="h-10">
   324→                    <SelectValue placeholder="Select nationality" />
   325→                  </SelectTrigger>
   326→                  <SelectContent>
   327→                    {NATIONALITY_OPTIONS.map((n) => (
   328→                      <SelectItem key={n.value} value={n.value}>
   329→                        {n.value} ({n.labelJa})
   330→                      </SelectItem>
   331→                    ))}
   332→                  </SelectContent>
   333→                </Select>
   334→              </Field>
   335→              <Field label="State / Place of origin (India)">
   336→                <Select
   337→                  value={data.placeOfOrigin ?? ""}
   338→                  onValueChange={(v) => update("placeOfOrigin", v)}
   339→                  disabled={data.nationality !== "India"}
   340→                >
   341→                  <SelectTrigger className="h-10">
   342→                    <SelectValue placeholder={data.nationality === "India" ? "Select state" : "Select nationality first"} />
   343→                  </SelectTrigger>
   344→                  <SelectContent>
   345→                    {INDIAN_STATES.map((s) => (
   346→                      <SelectItem key={s} value={s}>
   347→                        {s}
   348→                      </SelectItem>
   349→                    ))}
   350→                  </SelectContent>
   351→                </Select>
   352→              </Field>
   353→              <Field label="Address" className="sm:col-span-2">
   354→                <Input value={data.address ?? ""} onChange={(e) => update("address", e.target.value)} placeholder="Mangalagiri, Andhra Pradesh" />
   355→              </Field>
   356→              <Field label="Languages known" className="sm:col-span-2">
   357→                <div className="space-y-2">
   358→                  <div className="flex gap-2">
   359→                    <Select value={langInput} onValueChange={(v) => { addLanguage(v); }}>
   360→                      <SelectTrigger className="h-10">
   361→                        <SelectValue placeholder="Add a language…" />
   362→                      </SelectTrigger>
   363→                      <SelectContent>
   364→                        {LANGUAGE_OPTIONS.filter((l) => !data.languages.includes(l.value)).map((l) => (
   365→                          <SelectItem key={l.value} value={l.value}>
   366→                            {l.value} ({l.labelJa})
   367→                          </SelectItem>
   368→                        ))}
   369→                      </SelectContent>
   370→                    </Select>
   371→                  </div>
   372→                  {data.languages.length > 0 && (
   373→                    <div className="flex flex-wrap gap-2">
   374→                      {data.languages.map((lang, i) => (
   375→                        <Badge key={i} variant="secondary" className="px-3 py-1.5 gap-1.5">
   376→                          {lang}
   377→                          <span className="text-xs text-muted-foreground">
   378→                            {data.languagesJa[i]}
   379→                          </span>
   380→                          <button
   381→                            onClick={() => removeLanguage(i)}
   382→                            className="ml-1 hover:text-destructive"
   383→                            aria-label="Remove"
   384→                          >
   385→                            <X className="h-3 w-3" />
   386→                          </button>
   387→                        </Badge>
   388→                      ))}
   389→                    </div>
   390→                  )}
   391→                </div>
   392→              </Field>
   393→            </div>
   394→          </Section>
   395→
   396→          {/* Education */}
   397→          <Section icon={GraduationCap} title="Education" desc="Academic history" action={<AddButton onClick={addEducation} />}>
   398→            <div className="space-y-4">
   399→              {data.education.length === 0 && <EmptyHint text="No education entries yet. Click Add to start." />}
   400→              {data.education.map((edu, i) => (
   401→                <Card key={i} onRemove={() => removeEducation(i)}>
   402→                  <div className="grid sm:grid-cols-2 gap-3">
   403→                    <Field label="Year / Month (graduation)">
   404→                      <Input value={edu.year} onChange={(e) => updateEducation(i, { year: e.target.value })} placeholder="2026" />
   405→                    </Field>
   406→                    <Field label="Degree (English)">
   407→                      <Input value={edu.degree} onChange={(e) => updateEducation(i, { degree: e.target.value })} placeholder="B.Tech" />
   408→                    </Field>
   409→                    <Field label="Degree (Japanese)">
   410→                      <Input value={edu.degreeJa ?? ""} onChange={(e) => updateEducation(i, { degreeJa: e.target.value })} placeholder="理学士" />
   411→                    </Field>
   412→                    <Field label="Field (English)">
   413→                      <Input value={edu.field} onChange={(e) => updateEducation(i, { field: e.target.value })} placeholder="Computer Science" />
   414→                    </Field>
   415→                    <Field label="Field (Japanese)">
   416→                      <Input value={edu.fieldJa ?? ""} onChange={(e) => updateEducation(i, { fieldJa: e.target.value })} placeholder="コンピュータサイエンス" />
   417→                    </Field>
   418→                    <Field label="Institution (English)">
   419→                      <Input value={edu.institution} onChange={(e) => updateEducation(i, { institution: e.target.value })} placeholder="SRM University" />
   420→                    </Field>
   421→                    <Field label="Institution (Japanese)">
   422→                      <Input value={edu.institutionJa ?? ""} onChange={(e) => updateEducation(i, { institutionJa: e.target.value })} placeholder="SRM大学" />
   423→                    </Field>
   424→                  </div>
   425→                </Card>
   426→              ))}
   427→            </div>
   428→          </Section>
   429→
   430→          {/* Projects */}
   431→          <Section icon={Briefcase} title="Projects" desc="Your technical projects" action={<AddButton onClick={addProject} />}>
   432→            <div className="space-y-4">
   433→              {data.projects.length === 0 && <EmptyHint text="No projects yet. Click Add to start." />}
   434→              {data.projects.map((proj, i) => (
   435→                <Card key={i} onRemove={() => removeProject(i)}>
   436→                  <div className="grid sm:grid-cols-2 gap-3">
   437→                    <Field label="Period">
   438→                      <Input value={proj.period} onChange={(e) => updateProject(i, { period: e.target.value })} placeholder="2024 Sep – 2024 Dec" />
   439→                    </Field>
   440→                    <Field label="Project name (English)">
   441→                      <Input value={proj.name} onChange={(e) => updateProject(i, { name: e.target.value })} placeholder="Collaboration Learning Platform" />
   442→                    </Field>
   443→                    <Field label="Project name (Japanese)">
   444→                      <Input value={proj.nameJa ?? ""} onChange={(e) => updateProject(i, { nameJa: e.target.value })} placeholder="コラボ学習プラットフォーム" />
   445→                    </Field>
   446→                    <Field label="Tech stack">
   447→                      <Input value={proj.techStack ?? ""} onChange={(e) => updateProject(i, { techStack: e.target.value })} placeholder="React.js, Node.js, Firebase, Tailwind CSS" />
   448→                    </Field>
   449→                  </div>
   450→                  <Field label="Description (English)" className="mt-3">
   451→                    <Textarea rows={3} value={proj.description} onChange={(e) => updateProject(i, { description: e.target.value })} placeholder="Developed a real-time learning platform with chat, live sessions, and auth..." />
   452→                  </Field>
   453→                  <Field label="Description (Japanese)" className="mt-2">
   454→                    <Textarea rows={3} value={proj.descriptionJa ?? ""} onChange={(e) => updateProject(i, { descriptionJa: e.target.value })} placeholder="チャット機能やライブセッション、認証機能を実装したリアルタイム学習プラットフォーム..." />
   455→                  </Field>
   456→                </Card>
   457→              ))}
   458→            </div>
   459→          </Section>
   460→
   461→          {/* Activities */}
   462→          <Section icon={Heart} title="Activities / Clubs" desc="Extracurricular and volunteer work" action={<AddButton onClick={addActivity} />}>
   463→            <div className="space-y-4">
   464→              {data.activities.length === 0 && <EmptyHint text="No activities yet. Click Add to start." />}
   465→              {data.activities.map((act, i) => (
   466→                <Card key={i} onRemove={() => removeActivity(i)}>
   467→                  <div className="grid sm:grid-cols-2 gap-3">
   468→                    <Field label="Period">
   469→                      <Input value={act.period} onChange={(e) => updateActivity(i, { period: e.target.value })} placeholder="9 months" />
   470→                    </Field>
   471→                    <Field label="Duration (Japanese)">
   472→                      <Input value={act.duration ?? ""} onChange={(e) => updateActivity(i, { duration: e.target.value })} placeholder="9か月" />
   473→                    </Field>
   474→                    <Field label="Organization (English)">
   475→                      <Input value={act.organization} onChange={(e) => updateActivity(i, { organization: e.target.value })} placeholder="Wellness Club" />
   476→                    </Field>
   477→                    <Field label="Organization (Japanese)">
   478→                      <Input value={act.organizationJa ?? ""} onChange={(e) => updateActivity(i, { organizationJa: e.target.value })} placeholder="ウェルネスクラブ" />
   479→                    </Field>
   480→                    <Field label="Role (English)">
   481→                      <Input value={act.role} onChange={(e) => updateActivity(i, { role: e.target.value })} placeholder="Member" />
   482→                    </Field>
   483→                    <Field label="Role (Japanese)">
   484→                      <Input value={act.roleJa ?? ""} onChange={(e) => updateActivity(i, { roleJa: e.target.value })} placeholder="メンバー" />
   485→                    </Field>
   486→                  </div>
   487→                  <Field label="Duties (English)" className="mt-3">
   488→                    <Textarea rows={2} value={act.duties} onChange={(e) => updateActivity(i, { duties: e.target.value })} placeholder="Organized stress-relief activities and fitness workshops..." />
   489→                  </Field>
   490→                  <Field label="Duties (Japanese)" className="mt-2">
   491→                    <Textarea rows={2} value={act.dutiesJa ?? ""} onChange={(e) => updateActivity(i, { dutiesJa: e.target.value })} placeholder="ストレス解消活動を企画・推進..." />
   492→                  </Field>
   493→                </Card>
   494→              ))}
   495→            </div>
   496→          </Section>
   497→
   498→          {/* Awards */}
   499→          <Section icon={Award} title="Awards / Achievements" desc="Honors and accomplishments" action={<AddButton onClick={addAward} />}>
   500→            <div className="space-y-4">
   501→              {data.awards.length === 0 && <EmptyHint text="No awards yet. Click Add to start." />}
   502→              {data.awards.map((aw, i) => (
   503→                <Card key={i} onRemove={() => removeAward(i)}>
   504→                  <div className="grid sm:grid-cols-2 gap-3">

   507→                    </Field>
   508→                    <Field label="Title (English)">
   509→                      <Input value={aw.title} onChange={(e) => updateAward(i, { title: e.target.value })} placeholder="Research Day Participant" />
   510→                    </Field>
   511→                    <Field label="Title (Japanese)">
   512→                      <Input value={aw.titleJa ?? ""} onChange={(e) => updateAward(i, { titleJa: e.target.value })} placeholder="第9回リサーチデイ参加" />
   513→                    </Field>

   529→            </div>
   530→          </Section>
   531→
   532→          {/* Self-PR */}
   533→          <Section icon={Sparkles} title="Self-PR & Hobbies" desc="Your personal statement and interests">
   534→            <Field label="Self-PR (English)">

   536→            </Field>
   537→            <Field label="自己PR (Japanese)" className="mt-3">
   538→              <Textarea rows={5} value={data.selfPrJa ?? ""} onChange={(e) => update("selfPrJa", e.target.value)} placeholder="コンピュータサイエンスを専攻する学生です。フルスタック開発やAIプロジェクトに取り組む中で..." />
   539→            </Field>
   540→            <Field label="Hobbies (English)" className="mt-3">
   541→              <Input value={data.hobbies ?? ""} onChange={(e) => update("hobbies", e.target.value)} placeholder="Badminton, Fitness, Reading" />
   542→            </Field>
   543→            <Field label="趣味 (Japanese)" className="mt-2">
   544→              <Input value={data.hobbiesJa ?? ""} onChange={(e) => update("hobbiesJa", e.target.value)} placeholder="バドミントン、フィットネス、読書" />
   545→            </Field>
   546→          </Section>
   547→        </motion.div>
   548→      )}
   549→
   550→      {/* Previews */}
   551→      {tab === "preview-ja" && <ResumePreview data={data} lang="ja" />}
   552→      {tab === "preview-en" && <ResumePreview data={data} lang="en" />}
   553→    </div>
   554→  );
   555→}
