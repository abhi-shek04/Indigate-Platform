    65→function StatCard({
    66→  value,
    67→  suffix,
    68→  label,
    69→  delay,
    70→}: {
    71→  value: number;
    72→  suffix?: string;
    73→  label: string;
    74→  delay: number;
    75→}) {
    76→  const count = useCountUp(value, 1800);
    77→  const ref = useRef<HTMLDivElement>(null);
    78→  const [visible, setVisible] = useState(false);
    79→
    80→  useEffect(() => {
    81→    const el = ref.current;
    82→    if (!el) return;
    83→    const obs = new IntersectionObserver(
    84→      ([e]) => {
    85→        if (e.isIntersecting) {
    86→          setVisible(true);
    87→          obs.disconnect();
    88→        }
    89→      },
    90→      { threshold: 0.3 },
    91→    );
    92→    obs.observe(el);
    93→    return () => obs.disconnect();
    94→  }, []);
    95→
    96→  return (
    97→    <motion.div
    98→      ref={ref}
    99→      initial={{ opacity: 0, y: 20 }}
   100→      animate={visible ? { opacity: 1, y: 0 } : {}}
   101→      transition={{ duration: 0.6, ease: easeOutExpo, delay }}
   102→      className="text-center"
   103→    >
   104→      <div className="font-display text-4xl sm:text-5xl font-extrabold tracking-tight">
   105→        <span className="text-gradient-brand">
   106→          {count.toLocaleString()}
   107→          {suffix}
   108→        </span>
   109→      </div>
   110→      <p className="text-[12px] text-muted-foreground font-medium mt-1.5 uppercase tracking-wide">{label}</p>
   111→    </motion.div>
   112→  );
   113→}
