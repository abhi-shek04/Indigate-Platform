    58→function StatCard({
    59→  value,
    60→  suffix,
    61→  label,
    62→  delay,
    63→}: {
    64→  value: number;
    65→  suffix?: string;
    66→  label: string;
    67→  delay: number;
    68→}) {
    69→  const count = useCountUp(value, 1800);
    70→  const ref = useRef<HTMLDivElement>(null);
    71→  const [visible, setVisible] = useState(false);
    72→
    73→  useEffect(() => {
    74→    const el = ref.current;
    75→    if (!el) return;
    76→    const obs = new IntersectionObserver(
    77→      ([e]) => {
    78→        if (e.isIntersecting) {
    79→          setVisible(true);
    80→          obs.disconnect();
    81→        }
    82→      },
    83→      { threshold: 0.3 },
    84→    );
    85→    obs.observe(el);
    86→    return () => obs.disconnect();
    87→  }, []);
    88→
    89→  return (
    90→    <motion.div
    91→      ref={ref}
    92→      initial={{ opacity: 0, y: 20 }}
    93→      animate={visible ? { opacity: 1, y: 0 } : {}}
    94→      transition={{ duration: 0.6, ease: easeOutExpo, delay }}
    95→      className="text-center"
    96→    >
    97→      <div className="font-display text-4xl sm:text-5xl font-extrabold tracking-tight">
    98→        <span className="text-gradient-brand">
    99→          {count.toLocaleString()}
   100→          {suffix}
   101→        </span>
   102→      </div>
   103→      <p className="mt-1.5 text-sm font-medium text-muted-foreground">{label}</p>
   104→    </motion.div>
   105→  );
   106→}
