   227→   227→function useCompanyJobs() {
   228→   228→  const user = useApp((s) => s.user);
   229→   229→  const [jobs, setJobs] = useState<JobDTO[] | null>(null);
   230→   230→  const [loading, setLoading] = useState(true);
   231→   231→
   232→   232→  const load = useCallback(async () => {
   233→   233→    setLoading(true);
   234→   234→    try {
   235→   235→      const res = await api<{ jobs: JobDTO[] }>(`/api/jobs?limit=50`);
   236→   236→      // Filter to this company's jobs (including inactive ones we own)
   237→   237→      const mine = (res.jobs || []).filter(
   238→   238→        (j) => j.company?.userId === user?.id,
   239→   239→      );
   240→   240→      setJobs(mine);
   241→   241→    } catch {
   242→   242→      setJobs([]);
   243→   243→    } finally {
   244→   244→      setLoading(false);
   245→   245→    }
   246→   246→  }, [user?.id]);
   247→   247→
   248→   248→  useEffect(() => {
   249→   249→    load();
   250→   250→  }, [load]);
   251→   251→
   252→   252→  return { jobs, loading, reload: load };
   253→   253→}

   255→   255→function useCompanyApps() {
   256→   256→  const [apps, setApps] = useState<ApplicationDTO[] | null>(null);
   257→   257→  const [loading, setLoading] = useState(true);
   258→   258→
   259→   259→  const load = useCallback(async () => {
   260→   260→    setLoading(true);
   261→   261→    try {
   262→   262→      const res = await api<{ applications: ApplicationDTO[] }>(
   263→   263→        "/api/applications",
   264→   264→      );
   265→   265→      setApps(res.applications);
   266→   266→    } catch {
   267→   267→      setApps([]);
   268→   268→    } finally {
   269→   269→      setLoading(false);
   270→   270→    }
   271→   271→  }, []);
   272→   272→
   273→   273→  useEffect(() => {
   274→   274→    load();
   275→   275→  }, [load]);
   276→   276→
   277→   277→  return { apps, loading, reload: load };
