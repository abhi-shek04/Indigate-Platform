    58→function StatCard({
    59→  value,
    60→  suffix,
    61→  label,
    62→  delay,
    63→  icon: Icon,
    64→}: {
    65→  value: number;
    66→  suffix?: string;
    67→  label: string;
    68→  delay: number;
    69→  icon: React.ComponentType<{ className?: string }>;
    70→}) {
    71→  const count = useCountUp(value, 1800);
    72→  const ref = useRef<HTMLDivElement>(null);
    73→  const [visible, setVisible] = useState(false);
    74→
    75→  useEffect(() => {
    76→    const el = ref.current;
    77→    if (!el) return;
    78→    const obs = new IntersectionObserver(
    79→      ([e]) => {
    80→        if (e.isIntersecting) {
    81→          setVisible(true);
    82→          obs.disconnect();
    83→        }
    84→      },
    85→      { threshold: 0.3 },
    86→    );
    87→    obs.observe(el);
    88→    return () => obs.disconnect();
    89→  }, []);
    90→
    91→  return (
    92→    <motion.div
    93→      ref={ref}
    94→      initial={{ opacity: 0, y: 24 }}
    95→      animate={visible ? { opacity: 1, y: 0 } : {}}
    96→      transition={{ duration: 0.6, ease: easeOutExpo, delay }}
    97→      whileHover={{ y: -4 }}
    98→      className="group relative rounded-2xl border border-border bg-card/80 glass px-5 py-6 text-center transition-all hover:border-saffron/40 hover:shadow-premium"
    99→    >
   100→      {/* Saffron accent line on top */}
   101→      <span className="absolute inset-x-8 top-0 h-px bg-gradient-to-r from-transparent via-saffron/60 to-transparent opacity-70 group-hover:opacity-100 transition-opacity" />
   102→      <div className="relative flex flex-col items-center">
   103→        <motion.div
   104→          whileHover={{ rotate: [0, -8, 8, 0], scale: 1.08 }}
   105→          transition={{ duration: 0.5 }}
   106→          className="mb-3 grid place-items-center h-11 w-11 rounded-xl bg-saffron/10 text-saffron ring-1 ring-inset ring-saffron/20"
   107→        >
   108→          <Icon className="h-5 w-5" />
   109→        </motion.div>
   110→        <div className="font-display text-4xl sm:text-5xl font-extrabold tracking-tight leading-none">
   111→          <span className="text-gradient-brand">
   112→            {count.toLocaleString()}
   113→            {suffix}
   114→          </span>
   115→        </div>
   116→        <p className="mt-2 text-xs sm:text-sm font-medium text-muted-foreground uppercase tracking-wide">
   117→          {label}
   118→        </p>
   119→      </div>
   120→    </motion.div>
   121→  );
   122→}
