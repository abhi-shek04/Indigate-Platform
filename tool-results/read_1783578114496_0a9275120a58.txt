    75→/**
    76→ * Returns a 0–100 percentage of how many resume sections have at least some
    77→ * data filled. Pure function — used to render the sidebar progress bar.
    78→ */
    79→function computeProgress(data: ResumeData): number {
    80→  const filled: boolean[] = [
    81→    !!(data.name?.trim() || data.email?.trim() || data.phone?.trim()),
    82→    data.education.length > 0,
    83→    data.projects.length > 0,
    84→    data.activities.length > 0,
    85→    data.awards.length > 0,
    86→    !!(data.selfPr?.trim() || data.selfPrJa?.trim() || data.hobbies?.trim()),
    87→  ];
    88→  return Math.round(
    89→    (filled.filter(Boolean).length / filled.length) * 100,
    90→  );
    91→}
