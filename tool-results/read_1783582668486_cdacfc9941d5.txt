   227→function useCompanyJobs() {
   228→  const user = useApp((s) => s.user);
   229→  const [jobs, setJobs] = useState<JobDTO[] | null>(null);
   230→  const [loading, setLoading] = useState(true);
   231→
   232→  const load = useCallback(async () => {
   233→    setLoading(true);
   234→    try {
   235→      const res = await api<{ jobs: JobDTO[] }>(`/api/jobs?limit=50`);
   236→      // Filter to this company's jobs (including inactive ones we own)
   237→      const mine = (res.jobs || []).filter(
   238→        (j) => j.company?.userId === user?.id,
   239→      );
   240→      setJobs(mine);
   241→    } catch {
   242→      setJobs([]);
   243→    } finally {
   244→      setLoading(false);
   245→    }
   246→  }, [user?.id]);
   247→
   248→  useEffect(() => {
   249→    load();
   250→  }, [load]);
   251→
   252→  return { jobs, loading, reload: load };
   253→}

   255→function useCompanyApps() {
   256→  const [apps, setApps] = useState<ApplicationDTO[] | null>(null);
   257→  const [loading, setLoading] = useState(true);
   258→
   259→  const load = useCallback(async () => {
   260→    setLoading(true);
   261→    try {
   262→      const res = await api<{ applications: ApplicationDTO[] }>(
   263→        "/api/applications",
   264→      );
   265→      setApps(res.applications);
   266→    } catch {
   267→      setApps([]);
   268→    } finally {
   269→      setLoading(false);
   270→    }
   271→  }, []);
   272→
   273→  useEffect(() => {
   274→    load();
   275→  }, [load]);
   276→
   277→  return { apps, loading, reload: load };
   278→}

   529→  async function toggleActive(j: JobDTO) {
   530→    setBusyId(j.id);
   531→    try {
   532→      await api(`/api/jobs/${j.id}`, {
   533→        method: "PUT",
   534→        body: JSON.stringify({ isActive: !j.isActive }),
   535→      });
   536→      toast.success(j.isActive ? "Job paused." : "Job reactivated.");
   537→      await reload();
   538→    } catch (e) {
   539→      toast.error(e instanceof Error ? e.message : "Failed to update.");
   540→    } finally {
   541→      setBusyId(null);
   542→    }
   543→  }

   548→      await api(`/api/jobs/${id}`, { method: "DELETE" });

   550→      await reload();
   551→    } catch (e) {
   552→      toast.error(e instanceof Error ? e.message : "Failed to delete.");
   553→    } finally {
   554→      setBusyId(null);
   555→      setDeleteId(null);
   556→    }
   557→  }
   558→
   559→  return (
   560→    <div className="space-y-4">
   561→      <div className="flex items-center justify-between gap-3">
   562→        <h2 className="font-display font-extrabold text-xl">

   578→        <EmptyState
   579→          icon={Briefcase}
   580→          title="No jobs yet"
   581→          description="Post your first role to start receiving applications."
   582→          action={
   583→            <Button
   584→              className="bg-brand-gradient text-white hover:opacity-90"
   585→              onClick={() => setTab("new")}
   586→            >
   587→              <Plus className="h-4 w-4" />
   588→              {t("dash.company.new")}

   600→                  <TableHead className="hidden md:table-cell">JLPT</TableHead>
