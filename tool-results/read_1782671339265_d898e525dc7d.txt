   133→STEP 3 — Create src/app/api/upload/route.ts:
   134→import { NextRequest } from 'next/server'
   135→import { getSession } from '@/lib/auth'
   136→import { ok, err, handleError } from '@/lib/api'
   137→import { db } from '@/lib/db'
   138→import { supabaseAdmin } from '@/lib/supabase'
   139→
   140→export async function POST(req: NextRequest) {
   141→  try {
   142→    const session = await getSession()
   143→    if (!session) return err('Unauthorized.', 401)
   144→
   145→    const formData = await req.formData()
   146→    const file = formData.get('file') as File | null
   147→    const kind = formData.get('kind') as string | null
   148→
   149→    if (!file || !kind) return err('Missing file or kind.', 400)
   150→
   151→    // Validate by kind
   152→    const validations: Record<string, { roles: string[]; types: string[]; maxMB: number }> = {
   153→      resume: { roles: ['CANDIDATE'], types: ['application/pdf'], maxMB: 5 },
   154→      logo:   { roles: ['COMPANY'],   types: ['image/jpeg','image/png','image/webp'], maxMB: 2 },
   155→      photo:  { roles: ['CANDIDATE'], types: ['image/jpeg','image/png','image/webp'], maxMB: 2 },
   156→    }
   157→    const rule = validations[kind]
   158→    if (!rule) return err('Invalid upload kind.', 400)
   159→    if (!rule.roles.includes(session.role)) return err('Forbidden.', 403)
   160→    if (!rule.types.includes(file.type)) return err(`Invalid file type for ${kind}.`, 400)
   161→    if (file.size > rule.maxMB * 1024 * 1024) return err(`File too large. Max ${rule.maxMB}MB.`, 400)
   162→
   163→    // Build storage path
   164→    const ext = file.name.split('.').pop() ?? 'bin'
   165→    const path = `${kind}s/${session.id}/${Date.now()}.${ext}`
   166→
   167→    // Convert File to Buffer
   168→    const buffer = Buffer.from(await file.arrayBuffer())
   169→
   170→    // Upload to Supabase Storage
   171→    const bucket = process.env.SUPABASE_STORAGE_BUCKET!
   172→    const { error: uploadError } = await supabaseAdmin.storage
   173→      .from(bucket)
   174→      .upload(path, buffer, { contentType: file.type, upsert: true })
   175→    if (uploadError) throw new Error(uploadError.message)
   176→
   177→    // Get public URL
   178→    const { data } = supabaseAdmin.storage.from(bucket).getPublicUrl(path)
   179→    const url = data.publicUrl
   180→
   181→    // Save URL to DB based on kind
   182→    if (kind === 'resume') {
   183→      await db.candidateProfile.update({
   184→        where: { userId: session.id },
   185→        data: { resumeUrl: url, resumeName: file.name },
   186→      })
   187→    } else if (kind === 'logo') {
   188→      await db.companyProfile.update({
   189→        where: { userId: session.id },
   190→        data: { logoUrl: url },
   191→      })
   192→    } else if (kind === 'photo') {
   193→      await db.candidateProfile.update({
   194→        where: { userId: session.id },
   195→        data: { photoUrl: url },
   196→      })
   197→    }
   198→
   199→    return ok({ url, name: file.name })
   200→  } catch (e) {
   201→    return handleError(e)
   202→  }
   203→}
