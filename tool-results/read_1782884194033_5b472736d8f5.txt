   123→   123→
   124→   124→  // Language tag management

   135→   135→      "languages",
   136→   136→      data.languages.filter((_, i) => i !== idx),
   137→   137→    );

   139→   139→      "languagesJa",
   140→   140→      data.languagesJa.filter((_, i) => i !== idx),
   141→   141→    );
   142→   142→  }
   143→   143→
   144→   144→  // Education

   147→   147→      ...data.education,

   149→   149→    ]);

   161→   161→      ...data.projects,
   162→   162→      { period: "", name: "", description: "", techStack: "" },
   163→   163→    ]);
   164→   164→  }

   166→   166→    update("projects", data.projects.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   167→   167→  }
   168→   168→  function removeProject(i: number) {
   169→   169→    update("projects", data.projects.filter((_, j) => j !== i));
   170→   170→  }
   171→   171→

   174→   174→    update("activities", [
   175→   175→      ...data.activities,
   176→   176→      { period: "", organization: "", role: "", duties: "" },

   180→   180→    update("activities", data.activities.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   181→   181→  }
   182→   182→  function removeActivity(i: number) {
   183→   183→    update("activities", data.activities.filter((_, j) => j !== i));
   184→   184→  }
   185→   185→
   186→   186→  // Awards

   188→   188→    update("awards", [
   189→   189→      ...data.awards,
   190→   190→      { year: "", title: "", description: "", organization: "" },
   191→   191→    ]);
   192→   192→  }

   194→   194→    update("awards", data.awards.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   195→   195→  }

   197→   197→    update("awards", data.awards.filter((_, j) => j !== i));
   198→   198→  }

   221→   221→            <FileText className="h-7 w-7 text-saffron" />

   239→   239→                <Download className="h-4 w-4 mr-1.5" />

   249→   249→              <Button variant="outline" size="sm" disabled={loading} className="font-semibold h-9">
   250→   250→                <Download className="h-4 w-4 mr-1.5" />

   280→   280→            {tb.label}
   281→   281→          </button>

   283→   283→      </div>

   286→   286→      {tab === "edit" && (
   287→   287→        <motion.div

   290→   290→          className="print:hidden space-y-6"
   291→   291→        >
   292→   292→          {/* Personal Info */}
   293→   293→          <Section icon={User} title="Personal Information" desc="Your basic details — selectable fields make it faster">
   294→   294→            <div className="grid sm:grid-cols-2 gap-4">
   295→   295→              <Field label="Full name (English)">
   296→   296→                <Input value={data.name} onChange={(e) => update("name", e.target.value)} placeholder="Abhishek" />
   297→   297→              </Field>
   298→   298→              <Field label="氏名 (Katakana reading)">
   299→   299→                <Input value={data.nameJa ?? ""} onChange={(e) => update("nameJa", e.target.value)} placeholder="アビシェーク" />
   300→   300→              </Field>
   301→   301→              <Field label="Date of birth">
   302→   302→                <Input type="date" value={data.dob ?? ""} onChange={(e) => update("dob", e.target.value)} />
   303→   303→              </Field>
   304→   304→              <Field label="Gender">
   305→   305→                <Select value={data.gender ?? ""} onValueChange={(v) => update("gender", v)}>
   306→   306→                  <SelectTrigger className="h-10">
   307→   307→                    <SelectValue placeholder="Select gender" />
   308→   308→                  </SelectTrigger>
   309→   309→                  <SelectContent>
   310→   310→                    {GENDER_OPTIONS.map((g) => (

   332→   332→                        {n.value} ({n.labelJa})

   350→   350→                        {s}

   354→   354→                </Select>
   355→   355→              </Field>
   356→   356→              <Field label="Address" className="sm:col-span-2">

   389→   389→                          </button>
