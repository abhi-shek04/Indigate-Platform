   217→    subject,
   218→    html,
   219→  })
   220→  if (error) console.error('[EMAIL ERROR]', error)

   795→
   796→const store = new Map<string, { count: number; resetAt: number }>()
   797→
   798→export function rateLimit(key: string, maxRequests: number, windowMs: number): boolean {
   799→  const now = Date.now()
   800→  const record = store.get(key)
   801→  if (!record || now > record.resetAt) {
   802→    store.set(key, { count: 1, resetAt: now + windowMs })
   803→    return true // allowed
   804→  }
   805→  if (record.count >= maxRequests) return false // blocked
   806→  record.count++
   807→  return true // allowed
   808→}
   809→
   810→// Clean up old entries every 5 minutes
   811→setInterval(() => {
   812→  const now = Date.now()
   813→  for (const [key, val] of store.entries()) {
   814→    if (now > val.resetAt) store.delete(key)
   815→  }
   816→}, 5 * 60 * 1000)
