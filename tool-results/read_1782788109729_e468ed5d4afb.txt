   170→   170→function computeCompletion(c: CandidateProfileDTO | null): number {
   171→   171→  if (!c) return 0;
   172→   172→  const checks: boolean[] = [
   173→   173→    !!c.fullName?.trim(),
   174→   174→    !!c.phone?.trim(),
   175→   175→    !!c.bio?.trim(),
   176→   176→    !!c.location?.trim(),
   177→   177→    c.jlptLevel !== "NONE",
   178→   178→    (c.skills?.length ?? 0) >= 3,
   179→   179→    !!c.resumeUrl,
   180→   180→    c.experienceYears > 0,
   181→   181→    !!c.education && c.education.length > 0,
   182→   182→    !!c.linkedinUrl?.trim(),
   183→   183→  ];
   184→   184→  const filled = checks.filter(Boolean).length;
   185→   185→  return Math.round((filled / checks.length) * 100);
   186→   186→}

   411→   411→      setApps([]);

   418→   418→    load();

   425→   425→        method: "PATCH",
   426→   426→        body: JSON.stringify({ status: "WITHDRAWN" }),
   427→   427→      });

   429→   429→      await load();
   430→   430→      await refreshAuth();
   431→   431→    } catch (e) {
   432→   432→      toast.error(e instanceof Error ? e.message : "Failed to withdraw.");
   433→   433→    } finally {
   434→   434→      setBusyId(null);
   435→   435→    }
   436→   436→  }
   437→   437→

   439→   439→    return (
   440→   440→      <div className="space-y-4">

   442→   442→      </div>
   443→   443→    );

   458→   458→            {t("dash.apps.apply")}
   459→   459→          </Button>
   460→   460→        }
   461→   461→      />
   462→   462→    );
   463→   463→  }
   464→   464→
   465→   465→  return (
   466→   466→    <div className="space-y-4">
   467→   467→      <SectionCard
   468→   468→        title={`${apps.length} application${apps.length === 1 ? "" : "s"}`}
   469→   469→        bodyClassName="p-0"
   470→   470→      >
   471→   471→        <div className="max-h-[70vh] overflow-y-auto scroll-area">
   472→   472→          <Table>
   473→   473→            <TableHeader className="sticky top-0 bg-card z-10">
   474→   474→              <TableRow>
   475→   475→                <TableHead className="pl-5 sm:pl-6">Role</TableHead>

   479→   479→              </TableRow>
   480→   480→            </TableHeader>
   481→   481→            <TableBody>
   482→   482→              {apps.map((a) => (
   483→   483→                <Fragment key={a.id}>
   484→   484→                <TableRow>
   485→   485→                  <TableCell className="pl-5 sm:pl-6">
   486→   486→                    <button
   487→   487→                      onClick={() =>
   488→   488→                        a.job && navigate("job-detail", { jobId: a.job.id })
   489→   489→                      }
   490→   490→                      className="text-left"
   491→   491→                    >
   492→   492→                      <p className="font-semibold text-sm hover:text-crimson transition-colors">
   493→   493→                        {a.job?.title ?? "Job removed"}
   494→   494→                      </p>
   495→   495→                      <p className="text-xs text-muted-foreground">
   496→   496→                        {a.job?.company?.companyName} ·{" "}
   497→   497→                        {a.job?.location}
   498→   498→                      </p>
   499→   499→                    </button>
   500→   500→                  </TableCell>
   501→   501→                  <TableCell className="hidden sm:table-cell text-sm text-muted-foreground">
   502→   502→                    {formatDate(a.appliedAt, locale)}
   503→   503→                  </TableCell>
   504→   504→                  <TableCell>
   505→   505→                    <Badge
   506→   506→                      variant="outline"
   507→   507→                      className={cn("font-semibold", STATUS_BADGE[a.status])}
   508→   508→                    >
   509→   509→                      {t(`status.${a.status}`)}
   510→   510→                    </Badge>
   511→   511→                  </TableCell>
   512→   512→                  <TableCell className="text-right pr-5 sm:pr-6">

   878→   878→  async function uploadFile(file: File) {

   880→   880→      toast.error("Resume must be a PDF.");

   890→   890→      const fd = new FormData();
   891→   891→      fd.append("file", file);
   892→   892→      fd.append("kind", "resume");
   893→   893→      // simulate upload progress (fetch doesn't give us easy progress)
   894→   894→      const fakeTimer = setInterval(() => {
   895→   895→        setProgress((p) => Math.min(90, (p ?? 0) + 10));
   896→   896→      }, 80);
   897→   897→      const res = await fetch("/api/upload", { method: "POST", body: fd });
   898→   898→      clearInterval(fakeTimer);
   899→   899→      setProgress(100);
   900→   900→      if (!res.ok) {
   901→   901→        const data = await res.json().catch(() => null);
   902→   902→        throw new Error(data?.error || "Upload failed.");

   904→   904→      toast.success("Resume uploaded.");

   907→   907→      toast.error(e instanceof Error ? e.message : "Upload failed.");
   908→   908→    } finally {
   909→   909→      setBusy(false);
