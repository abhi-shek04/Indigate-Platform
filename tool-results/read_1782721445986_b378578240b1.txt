    63→function StatCard({
    64→  value,
    65→  suffix,
    66→  label,
    67→  delay,
    68→}: {
    69→  value: number;
    70→  suffix?: string;
    71→  label: string;
    72→  delay: number;
    73→}) {
    74→  const count = useCountUp(value, 1800);
    75→  const ref = useRef<HTMLDivElement>(null);
    76→  const [visible, setVisible] = useState(false);
    77→
    78→  useEffect(() => {
    79→    const el = ref.current;
    80→    if (!el) return;
    81→    const obs = new IntersectionObserver(
    82→      ([e]) => {
    83→        if (e.isIntersecting) {
    84→          setVisible(true);
    85→          obs.disconnect();
    86→        }
    87→      },
    88→      { threshold: 0.3 },
    89→    );
    90→    obs.observe(el);
    91→    return () => obs.disconnect();
    92→  }, []);
    93→
    94→  return (
    95→    <motion.div
    96→      ref={ref}
    97→      initial={{ opacity: 0, y: 20 }}
    98→      animate={visible ? { opacity: 1, y: 0 } : {}}
    99→      transition={{ duration: 0.6, ease: easeOutExpo, delay }}
   100→      className="text-center"
   101→    >
   102→      <div className="font-display text-4xl sm:text-5xl font-extrabold tracking-tight">
   103→        <span className="text-gradient-brand">
   104→          {count.toLocaleString()}
   105→          {suffix}
   106→        </span>
   107→      </div>
   108→      <p className="mt-1.5 text-sm font-medium text-muted-foreground">{label}</p>
   109→    </motion.div>
   110→  );
   111→}
