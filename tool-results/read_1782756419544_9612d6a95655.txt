   551→function JobsTab() {
   552→  const { t, locale } = useT();
   553→  const [items, setItems] = useState<JobDTO[] | null>(null);
   554→  const [loading, setLoading] = useState(true);
   555→  const [search, setSearch] = useState("");
   556→  const [busyId, setBusyId] = useState<string | null>(null);
   557→  const [deleteId, setDeleteId] = useState<string | null>(null);
   558→  const [editorState, setEditorState] = useState<{
   559→    mode: "create" | "edit";
   560→    job: JobDTO | null;
   561→  } | null>(null);
   562→
   563→  const load = useCallback(async () => {
   564→    setLoading(true);
   565→    try {
   566→      const res = await api<{ items: JobDTO[] }>("/api/admin/list/jobs");
   567→      setItems(res.items);
   568→    } catch {
   569→      setItems([]);
   570→    } finally {
   571→      setLoading(false);
   572→    }
   573→  }, []);
   574→
   575→  useEffect(() => {
   576→    load();
   577→  }, [load]);
   578→
   579→  const filtered = useMemo(() => {
   580→    if (!items) return [];
   581→    const q = search.trim().toLowerCase();
   582→    if (!q) return items;
   583→    return items.filter(
   584→      (j) =>
   585→        j.title.toLowerCase().includes(q) ||
   586→        j.company.companyName.toLowerCase().includes(q) ||
   587→        j.location.toLowerCase().includes(q),
   588→    );
   589→  }, [items, search]);
   590→
   591→  async function toggleActive(j: JobDTO) {
   592→    setBusyId(j.id);
   593→    try {
   594→      await api(`/api/jobs/${j.id}`, {
   595→        method: "PUT",
