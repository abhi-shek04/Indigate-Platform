   572→function JobsTab() {
   573→  const { t, locale } = useT();
   574→  const [items, setItems] = useState<JobDTO[] | null>(null);
   575→  const [loading, setLoading] = useState(true);
   576→  const [search, setSearch] = useState("");
   577→  const [busyId, setBusyId] = useState<string | null>(null);
   578→  const [deleteId, setDeleteId] = useState<string | null>(null);
   579→  const [editorState, setEditorState] = useState<{
   580→    mode: "create" | "edit";
   581→    job: JobDTO | null;
   582→  } | null>(null);
   583→
   584→  const load = useCallback(async () => {
   585→    setLoading(true);
   586→    try {
   587→      const res = await api<{ items: JobDTO[] }>("/api/admin/list/jobs");
   588→      setItems(res.items);
   589→    } catch {
   590→      setItems([]);
   591→    } finally {
   592→      setLoading(false);
   593→    }
   594→  }, []);
   595→
   596→  useEffect(() => {
   597→    load();
   598→  }, [load]);
   599→
   600→  const filtered = useMemo(() => {
   601→    if (!items) return [];
   602→    const q = search.trim().toLowerCase();
   603→    if (!q) return items;
   604→    return items.filter(
   605→      (j) =>
   606→        j.title.toLowerCase().includes(q) ||
   607→        j.company.companyName.toLowerCase().includes(q) ||
   608→        j.location.toLowerCase().includes(q),
   609→    );
   610→  }, [items, search]);
   611→
   612→  async function toggleActive(j: JobDTO) {
   613→    setBusyId(j.id);
   614→    try {
   615→      await api(`/api/jobs/${j.id}`, {
   616→        method: "PUT",
   617→        body: JSON.stringify({ isActive: !j.isActive }),
   618→      });
   619→      toast.success(j.isActive ? "Paused." : "Activated.");
   620→      await load();
   621→    } catch (e) {
   622→      toast.error(e instanceof Error ? e.message : "Failed.");
   623→    } finally {
   624→      setBusyId(null);
   625→    }
   626→  }
   627→
   628→  async function remove(id: string) {
   629→    setBusyId(id);
   630→    try {
   631→      await api(`/api/jobs/${id}`, { method: "DELETE" });
   632→      toast.success("Job deleted.");
   633→      await load();
   634→    } catch (e) {
   635→      toast.error(e instanceof Error ? e.message : "Failed.");
   636→    } finally {
   637→      setBusyId(null);
   638→      setDeleteId(null);
   639→    }
   640→  }
   641→
