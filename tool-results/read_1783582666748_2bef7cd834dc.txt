   578→function JobsTab() {
   579→  const { t, locale } = useT();
   580→  const [items, setItems] = useState<JobDTO[] | null>(null);
   581→  const [loading, setLoading] = useState(true);
   582→  const [search, setSearch] = useState("");
   583→  const [busyId, setBusyId] = useState<string | null>(null);
   584→  const [deleteId, setDeleteId] = useState<string | null>(null);
   585→  const [editorState, setEditorState] = useState<{
   586→    mode: "create" | "edit";
   587→    job: JobDTO | null;
   588→  } | null>(null);
   589→
   590→  const load = useCallback(async () => {
   591→    setLoading(true);
   592→    try {
   593→      const res = await api<{ items: JobDTO[] }>("/api/admin/list/jobs");
   594→      setItems(res.items);
   595→    } catch {
   596→      setItems([]);
   597→    } finally {
   598→      setLoading(false);
   599→    }
   600→  }, []);
   601→
   602→  useEffect(() => {
   603→    load();
   604→  }, [load]);
   605→
   606→  const filtered = useMemo(() => {
   607→    if (!items) return [];
   608→    const q = search.trim().toLowerCase();
   609→    if (!q) return items;
   610→    return items.filter(
   611→      (j) =>
   612→        j.title.toLowerCase().includes(q) ||
   613→        j.company.companyName.toLowerCase().includes(q) ||
   614→        j.location.toLowerCase().includes(q),
   615→    );
   616→  }, [items, search]);
   617→
   618→  async function toggleActive(j: JobDTO) {
   619→    setBusyId(j.id);
   620→    try {
   621→      await api(`/api/jobs/${j.id}`, {
   622→        method: "PUT",
   623→        body: JSON.stringify({ isActive: !j.isActive }),
   624→      });
   625→      toast.success(j.isActive ? "Paused." : "Activated.");
   626→      await load();
   627→    } catch (e) {
   628→      toast.error(e instanceof Error ? e.message : "Failed.");
   629→    } finally {
   630→      setBusyId(null);
   631→    }
   632→  }
   633→
   634→  async function remove(id: string) {
   635→    setBusyId(id);
   636→    try {
   637→      await api(`/api/jobs/${id}`, { method: "DELETE" });
