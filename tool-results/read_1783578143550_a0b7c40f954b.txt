    75→    75→/**
    76→    76→ * Returns a 0–100 percentage of how many resume sections have at least some
    77→    77→ * data filled. Pure function — used to render the sidebar progress bar.
    78→    78→ */
    79→    79→function computeProgress(data: ResumeData): number {
    80→    80→  const filled: boolean[] = [
    81→    81→    !!(data.name?.trim() || data.email?.trim() || data.phone?.trim()),
    82→    82→    data.education.length > 0,
    83→    83→    data.projects.length > 0,
    84→    84→    data.activities.length > 0,
    85→    85→    data.awards.length > 0,
    86→    86→    !!(data.selfPr?.trim() || data.selfPrJa?.trim() || data.hobbies?.trim()),
    87→    87→  ];
    88→    88→  return Math.round(
    89→    89→    (filled.filter(Boolean).length / filled.length) * 100,
    90→    90→  );
    91→    91→}
