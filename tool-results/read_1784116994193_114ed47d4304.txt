    94→function StatCard({
    95→  value,
    96→  suffix,
    97→  label,
    98→  delay,
    99→  icon: Icon,
   100→}: {
   101→  value: number;
   102→  suffix?: string;
   103→  label: string;
   104→  delay: number;
   105→  icon: React.ComponentType<{ className?: string }>;
   106→}) {
   107→  const count = useCountUp(value, 1800);
   108→  const ref = useRef<HTMLDivElement>(null);
   109→  const [visible, setVisible] = useState(false);
   110→
   111→  useEffect(() => {
   112→    const el = ref.current;
   113→    if (!el) return;
   114→    const obs = new IntersectionObserver(
   115→      ([e]) => {
   116→        if (e.isIntersecting) {
   117→          setVisible(true);
   118→          obs.disconnect();
   119→        }
   120→      },
   121→      { threshold: 0.3 },
   122→    );
   123→    obs.observe(el);
   124→    return () => obs.disconnect();
   125→  }, []);
   126→
   127→  return (
   128→    <motion.div
   129→      ref={ref}
   130→      initial={{ opacity: 0, y: 24 }}
   131→      animate={visible ? { opacity: 1, y: 0 } : {}}
   132→      transition={{ duration: 0.6, ease: easeOutExpo, delay }}
   133→      whileHover={{ y: -4 }}
   134→      className="hero-stat group px-5 py-6 text-center transition-transform hover:-translate-y-1"
   135→    >
   136→      {/* Saffron accent line on top is provided by .hero-stat::after */}
   137→      <div className="relative flex flex-col items-center">
   138→        <motion.div
   139→          whileHover={{ rotate: [0, -8, 8, 0], scale: 1.08 }}
   140→          transition={{ duration: 0.5 }}
   141→          className="mb-3 grid place-items-center h-11 w-11 rounded-xl bg-saffron/10 text-saffron ring-1 ring-inset ring-saffron/20"
   142→        >
   143→          <Icon className="h-5 w-5" />
   144→        </motion.div>
   145→        <div className="font-display text-4xl sm:text-5xl font-extrabold tracking-tight leading-none">
   146→          <span className="text-gradient-brand">
   147→            {count.toLocaleString()}
   148→            {suffix}
   149→          </span>
   150→        </div>
   151→        <p className="mt-2 text-xs sm:text-sm font-medium text-muted-foreground uppercase tracking-wide">
   152→          {label}
   153→        </p>
   154→      </div>
   155→    </motion.div>
   156→  );
   157→}

   161→  stage,
   162→  count,
   163→  pct,
   164→  delay,
   165→}: {
   166→  stage: string;
   167→  count: string;
   168→  pct: number;
   169→  delay: number;
   170→}) {
   171→  const ref = useRef<HTMLDivElement>(null);
   172→  const [visible, setVisible] = useState(false);
   173→  useEffect(() => {
   174→    const el = ref.current;
   175→    if (!el) return;
   176→    const obs = new IntersectionObserver(
   177→      ([e]) => {
   178→        if (e.isIntersecting) {
   179→          setVisible(true);
   180→          obs.disconnect();
   181→        }
   182→      },
   183→      { threshold: 0.4 },
   184→    );
   185→    obs.observe(el);
   186→    return () => obs.disconnect();
   187→  }, []);
   188→
   189→  return (
   190→    <div ref={ref}>
   191→      <div className="flex justify-between text-xs mb-1.5">
   192→        <span className="font-medium">{stage}</span>
   193→        <span className="text-muted-foreground">{count}</span>

   199→          animate={visible ? { width: `${pct}%` } : {}}
   200→          transition={{ duration: 1.1, ease: easeOutExpo, delay }}
   201→        />
   202→      </div>
   203→    </div>
   204→  );
