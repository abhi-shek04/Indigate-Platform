    94→    94→function StatCard({
    95→    95→  value,
    96→    96→  suffix,
    97→    97→  label,
    98→    98→  delay,
    99→    99→  icon: Icon,
   100→   100→}: {
   101→   101→  value: number;
   102→   102→  suffix?: string;
   103→   103→  label: string;
   104→   104→  delay: number;
   105→   105→  icon: React.ComponentType<{ className?: string }>;
   106→   106→}) {
   107→   107→  const count = useCountUp(value, 1800);
   108→   108→  const ref = useRef<HTMLDivElement>(null);
   109→   109→  const [visible, setVisible] = useState(false);
   110→   110→
   111→   111→  useEffect(() => {
   112→   112→    const el = ref.current;
   113→   113→    if (!el) return;
   114→   114→    const obs = new IntersectionObserver(
   115→   115→      ([e]) => {
   116→   116→        if (e.isIntersecting) {
   117→   117→          setVisible(true);
   118→   118→          obs.disconnect();
   119→   119→        }
   120→   120→      },
   121→   121→      { threshold: 0.3 },
   122→   122→    );
   123→   123→    obs.observe(el);
   124→   124→    return () => obs.disconnect();
   125→   125→  }, []);
   126→   126→
   127→   127→  return (
   128→   128→    <motion.div
   129→   129→      ref={ref}
   130→   130→      initial={{ opacity: 0, y: 24 }}
   131→   131→      animate={visible ? { opacity: 1, y: 0 } : {}}
   132→   132→      transition={{ duration: 0.6, ease: easeOutExpo, delay }}
   133→   133→      whileHover={{ y: -4 }}
   134→   134→      className="hero-stat group px-5 py-6 text-center transition-transform hover:-translate-y-1"
   135→   135→    >
   136→   136→      {/* Saffron accent line on top is provided by .hero-stat::after */}
   137→   137→      <div className="relative flex flex-col items-center">
   138→   138→        <motion.div
   139→   139→          whileHover={{ rotate: [0, -8, 8, 0], scale: 1.08 }}
   140→   140→          transition={{ duration: 0.5 }}
   141→   141→          className="mb-3 grid place-items-center h-11 w-11 rounded-xl bg-saffron/10 text-saffron ring-1 ring-inset ring-saffron/20"
   142→   142→        >
   143→   143→          <Icon className="h-5 w-5" />
   144→   144→        </motion.div>
   145→   145→        <div className="font-display text-4xl sm:text-5xl font-extrabold tracking-tight leading-none">
   146→   146→          <span className="text-gradient-brand">
   147→   147→            {count.toLocaleString()}
   148→   148→            {suffix}
   149→   149→          </span>
   150→   150→        </div>
   151→   151→        <p className="mt-2 text-xs sm:text-sm font-medium text-muted-foreground uppercase tracking-wide">
   152→   152→          {label}
   153→   153→        </p>
   154→   154→      </div>
   155→   155→    </motion.div>
   156→   156→  );
   157→   157→}
