     1→"use client";
     2→
     3→import { useEffect, useState, useCallback } from "react";
     4→import { useApp } from "@/lib/store";
     5→import { useT } from "@/lib/use-t";
     6→import { api } from "@/lib/api-client";
     7→import { Button } from "@/components/ui/button";
     8→import { Input } from "@/components/ui/input";
     9→import { Label } from "@/components/ui/label";
    10→import { Textarea } from "@/components/ui/textarea";
    11→import { Badge } from "@/components/ui/badge";
    12→import {
    13→  Select,
    14→  SelectContent,
    15→  SelectItem,
    16→  SelectTrigger,
    17→  SelectValue,
    18→} from "@/components/ui/select";
    19→import {
    20→  ArrowLeft,
    21→  Plus,
    22→  Trash2,
    23→  Save,
    24→  Printer,
    25→  FileText,
    26→  User,
    27→  GraduationCap,
    28→  Briefcase,
    29→  Award,
    30→  Heart,
    31→  Globe,
    32→  Sparkles,
    33→  Loader2,
    34→  X,
    35→  Download,
    36→} from "lucide-react";
    37→import { motion } from "framer-motion";
    38→import { toast } from "sonner";
    39→import { cn } from "@/lib/utils";
    40→import { PDFDownloadLink } from "@react-pdf/renderer";
    41→import { EnglishResumePDF } from "@/lib/pdf-templates/english-resume-pdf";
    42→import { JapaneseResumePDF } from "@/lib/pdf-templates/japanese-resume-pdf";
    43→import {
    44→  EMPTY_RESUME,
    45→  GENDER_OPTIONS,
    46→  NATIONALITY_OPTIONS,
    47→  INDIAN_STATES,
    48→  LANGUAGE_OPTIONS,
    49→  STATE_JA,
    50→  type ResumeData,
    51→  type ResumeEducation,
    52→  type ResumeProject,
    53→  type ResumeActivity,
    54→  type ResumeAward,
    55→} from "@/lib/resume-types";
    56→import { ResumePreview } from "./resume-preview";
    57→
    58→type Tab = "edit" | "preview-ja" | "preview-en";
    59→
    60→export function ResumeBuilder() {
    61→  const { t } = useT();
    62→  const navigate = useApp((s) => s.navigate);
    63→  const candidate = useApp((s) => s.candidate);
    64→  const user = useApp((s) => s.user);
    65→  const [data, setData] = useState<ResumeData>(EMPTY_RESUME);
    66→  const [loading, setLoading] = useState(true);
    67→  const [saving, setSaving] = useState(false);
    68→  const [tab, setTab] = useState<Tab>("edit");
    69→  const [langInput, setLangInput] = useState("");
    70→
    71→  const load = useCallback(async () => {
    72→    setLoading(true);
    73→    try {
    74→      const res = await api<{ resumeData: ResumeData | null }>(
    75→        "/api/candidates/me/resume",
    76→      );
    77→      if (res.resumeData) {
    78→        setData({ ...EMPTY_RESUME, ...res.resumeData });
    79→      } else if (candidate) {
    80→        setData({
    81→          ...EMPTY_RESUME,
    82→          name: candidate.fullName,
    83→          email: user?.email ?? "",
    84→          phone: candidate.phone ?? "",
    85→          nationality: "India",
    86→          languages: ["English"],
    87→          languagesJa: ["英語"],
    88→        });
    89→      }
    90→    } catch {
    91→      toast.error("Failed to load resume data.");
    92→    } finally {
    93→      setLoading(false);
    94→    }
    95→  }, [candidate?.id, user?.email]);
    96→
    97→  useEffect(() => {
    98→    load();
    99→  }, [load]);
   100→
   101→  async function save() {
   102→    setSaving(true);
   103→    try {
   104→      await api("/api/candidates/me/resume", {
   105→        method: "PUT",
   106→        body: JSON.stringify(data),
   107→      });
   108→      toast.success("Resume saved!");
   109→    } catch (e) {
   110→      toast.error(e instanceof Error ? e.message : "Failed to save.");
   111→    } finally {
   112→      setSaving(false);
   113→    }
   114→  }
   115→
   116→  function print() {
   117→    window.print();
   118→  }
   119→
   120→  function update<K extends keyof ResumeData>(key: K, value: ResumeData[K]) {
   121→    setData((prev) => ({ ...prev, [key]: value }));
   122→  }
   123→
   124→  // Language tag management
   125→  function addLanguage(lang: string) {
   126→    if (!lang.trim()) return;
   127→    const opt = LANGUAGE_OPTIONS.find((l) => l.value === lang);
   128→    if (data.languages.includes(lang)) return;
   129→    update("languages", [...data.languages, lang]);
   130→    update("languagesJa", [...data.languagesJa, opt?.labelJa ?? lang]);
   131→    setLangInput("");
   132→  }
   133→  function removeLanguage(idx: number) {
   134→    update(
   135→      "languages",
   136→      data.languages.filter((_, i) => i !== idx),
   137→    );
   138→    update(
   139→      "languagesJa",
   140→      data.languagesJa.filter((_, i) => i !== idx),
   141→    );
   142→  }
   143→
   144→  // Education
   145→  function addEducation() {
   146→    update("education", [
   147→      ...data.education,
   148→      { year: "", degree: "", field: "", institution: "" },
   149→    ]);
   150→  }
   151→  function updateEducation(i: number, patch: Partial<ResumeEducation>) {
   152→    update("education", data.education.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   153→  }
   154→  function removeEducation(i: number) {
   155→    update("education", data.education.filter((_, j) => j !== i));
   156→  }
   157→
   158→  // Projects
   159→  function addProject() {
   160→    update("projects", [
   161→      ...data.projects,
   162→      { period: "", name: "", description: "", techStack: "" },
   163→    ]);
   164→  }
   165→  function updateProject(i: number, patch: Partial<ResumeProject>) {
   166→    update("projects", data.projects.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   167→  }
   168→  function removeProject(i: number) {
   169→    update("projects", data.projects.filter((_, j) => j !== i));
   170→  }
   171→
   172→  // Activities
   173→  function addActivity() {
   174→    update("activities", [
   175→      ...data.activities,
   176→      { period: "", organization: "", role: "", duties: "" },
   177→    ]);
   178→  }
   179→  function updateActivity(i: number, patch: Partial<ResumeActivity>) {
   180→    update("activities", data.activities.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   181→  }
   182→  function removeActivity(i: number) {
   183→    update("activities", data.activities.filter((_, j) => j !== i));
   184→  }
   185→
   186→  // Awards
   187→  function addAward() {
   188→    update("awards", [
   189→      ...data.awards,
   190→      { year: "", title: "", description: "", organization: "" },
   191→    ]);
   192→  }
   193→  function updateAward(i: number, patch: Partial<ResumeAward>) {
   194→    update("awards", data.awards.map((e, j) => (j === i ? { ...e, ...patch } : e)));
   195→  }
   196→  function removeAward(i: number) {
   197→    update("awards", data.awards.filter((_, j) => j !== i));
   198→  }
   199→
   200→  if (loading) {
   201→    return (
   202→      <div className="flex-1 grid place-items-center py-20">
   203→        <Loader2 className="h-8 w-8 animate-spin text-saffron" />
   204→      </div>
   205→    );
   206→  }
   207→
   208→  return (
   209→    <div className="mx-auto max-w-5xl px-4 sm:px-6 lg:px-8 py-8 print:max-w-none print:p-0">
   210→      {/* Header */}
   211→      <div className="print:hidden flex items-center justify-between flex-wrap gap-4 mb-6">
   212→        <div>
   213→          <button
   214→            onClick={() => navigate("candidate")}
   215→            className="inline-flex items-center gap-1.5 text-sm font-medium text-muted-foreground hover:text-foreground mb-2"
   216→          >
   217→            <ArrowLeft className="h-4 w-4" />
   218→            {t("common.back")}
   219→          </button>
   220→          <h1 className="font-display text-2xl sm:text-3xl font-extrabold tracking-tight flex items-center gap-2">
   221→            <FileText className="h-7 w-7 text-saffron" />
   222→            Resume Builder
   223→          </h1>
   224→          <p className="mt-1 text-sm text-muted-foreground">
   225→            Create a professional resume in both English and Japanese 履歴書 format.
   226→          </p>
   227→        </div>
   228→        <div className="flex items-center gap-2 flex-wrap">
   229→          <Button onClick={save} disabled={saving} className="bg-brand-gradient text-white hover:opacity-90 font-semibold">
   230→            {saving ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Save className="mr-2 h-4 w-4" />}
   231→            Save
   232→          </Button>
   233→          <PDFDownloadLink
   234→            document={<EnglishResumePDF data={data} />}
   235→            fileName={`${data.name || "resume"}_EN.pdf`}
   236→          >
   237→            {({ loading }) => (
   238→              <Button variant="outline" size="sm" disabled={loading} className="font-semibold h-9">
   239→                <Download className="h-4 w-4 mr-1.5" />
   240→                {loading ? "Generating…" : "EN PDF"}
   241→              </Button>
   242→            )}
   243→          </PDFDownloadLink>
   244→          <PDFDownloadLink
   245→            document={<JapaneseResumePDF data={data} />}
   246→            fileName={`${data.nameJa || data.name || "resume"}_JP.pdf`}
   247→          >
   248→            {({ loading }) => (
   249→              <Button variant="outline" size="sm" disabled={loading} className="font-semibold h-9">
   250→                <Download className="h-4 w-4 mr-1.5" />
   251→                {loading ? "生成中…" : "履歴書 PDF"}
   252→              </Button>
   253→            )}
   254→          </PDFDownloadLink>
   255→          <Button onClick={print} variant="ghost" size="sm" className="font-semibold h-9">
   256→            <Printer className="h-4 w-4 mr-1.5" />
   257→            Print
   258→          </Button>
   259→        </div>
   260→      </div>
   261→
   262→      {/* Tabs */}
   263→      <div className="print:hidden flex items-center gap-1 mb-6 p-1 rounded-xl bg-muted w-fit">
   264→        {[
   265→          { key: "edit" as const, label: "Edit", icon: User },
   266→          { key: "preview-ja" as const, label: "日本語 履歴書", icon: Globe },
   267→          { key: "preview-en" as const, label: "English Resume", icon: FileText },
   268→        ].map((tb) => (
   269→          <button
   270→            key={tb.key}
   271→            onClick={() => setTab(tb.key)}
   272→            className={cn(
   273→              "inline-flex items-center gap-1.5 px-4 py-2 rounded-lg text-sm font-medium transition-colors",
   274→              tab === tb.key
   275→                ? "bg-card text-foreground shadow-premium"
   276→                : "text-muted-foreground hover:text-foreground",
   277→            )}
   278→          >
   279→            <tb.icon className="h-4 w-4" />
   280→            {tb.label}
   281→          </button>
   282→        ))}
   283→      </div>
   284→
   285→      {/* Edit form */}
   286→      {tab === "edit" && (
   287→        <motion.div
   288→          initial={{ opacity: 0, y: 8 }}
   289→          animate={{ opacity: 1, y: 0 }}
   290→          className="print:hidden space-y-6"
   291→        >
   292→          {/* Personal Info */}
   293→          <Section icon={User} title="Personal Information" desc="Your basic details — selectable fields make it faster">
   294→            <div className="grid sm:grid-cols-2 gap-4">
   295→              <Field label="Full name (English)">
   296→                <Input value={data.name} onChange={(e) => update("name", e.target.value)} placeholder="Abhishek" />
   297→              </Field>
   298→              <Field label="氏名 (Katakana reading)">
   299→                <Input value={data.nameJa ?? ""} onChange={(e) => update("nameJa", e.target.value)} placeholder="アビシェーク" />
   300→              </Field>
   301→              <Field label="Date of birth">
   302→                <Input type="date" value={data.dob ?? ""} onChange={(e) => update("dob", e.target.value)} />
   303→              </Field>
   304→              <Field label="Gender">
   305→                <Select value={data.gender ?? ""} onValueChange={(v) => update("gender", v)}>
   306→                  <SelectTrigger className="h-10">
   307→                    <SelectValue placeholder="Select gender" />
   308→                  </SelectTrigger>
   309→                  <SelectContent>
   310→                    {GENDER_OPTIONS.map((g) => (
   311→                      <SelectItem key={g.value} value={g.value}>
   312→                        {g.labelEn} ({g.labelJa})
   313→                      </SelectItem>
   314→                    ))}
   315→                  </SelectContent>
   316→                </Select>
   317→              </Field>
   318→              <Field label="Email">
   319→                <Input type="email" value={data.email} onChange={(e) => update("email", e.target.value)} placeholder="you@example.com" />
   320→              </Field>
   321→              <Field label="Phone">
   322→                <Input value={data.phone ?? ""} onChange={(e) => update("phone", e.target.value)} placeholder="+91 98765 43210" />
   323→              </Field>
   324→              <Field label="Nationality">
   325→                <Select value={data.nationality ?? ""} onValueChange={(v) => update("nationality", v)}>
   326→                  <SelectTrigger className="h-10">
   327→                    <SelectValue placeholder="Select nationality" />
   328→                  </SelectTrigger>
   329→                  <SelectContent>
   330→                    {NATIONALITY_OPTIONS.map((n) => (
   331→                      <SelectItem key={n.value} value={n.value}>
   332→                        {n.value} ({n.labelJa})
   333→                      </SelectItem>
   334→                    ))}
   335→                  </SelectContent>
   336→                </Select>
   337→              </Field>
   338→              <Field label="State / Place of origin (India)">
   339→                <Select
   340→                  value={data.placeOfOrigin ?? ""}
   341→                  onValueChange={(v) => update("placeOfOrigin", v)}
   342→                  disabled={data.nationality !== "India"}
   343→                >
   344→                  <SelectTrigger className="h-10">
   345→                    <SelectValue placeholder={data.nationality === "India" ? "Select state" : "Select nationality first"} />
   346→                  </SelectTrigger>
   347→                  <SelectContent>
   348→                    {INDIAN_STATES.map((s) => (
   349→                      <SelectItem key={s} value={s}>
   350→                        {s}
   351→                      </SelectItem>
   352→                    ))}
   353→                  </SelectContent>
   354→                </Select>
   355→              </Field>
   356→              <Field label="Address" className="sm:col-span-2">
   357→                <Input value={data.address ?? ""} onChange={(e) => update("address", e.target.value)} placeholder="Mangalagiri, Andhra Pradesh" />
   358→              </Field>
   359→              <Field label="Languages known" className="sm:col-span-2">
   360→                <div className="space-y-2">
   361→                  <div className="flex gap-2">
   362→                    <Select value={langInput} onValueChange={(v) => { addLanguage(v); }}>
   363→                      <SelectTrigger className="h-10">
   364→                        <SelectValue placeholder="Add a language…" />
   365→                      </SelectTrigger>
   366→                      <SelectContent>
   367→                        {LANGUAGE_OPTIONS.filter((l) => !data.languages.includes(l.value)).map((l) => (
   368→                          <SelectItem key={l.value} value={l.value}>
   369→                            {l.value} ({l.labelJa})
   370→                          </SelectItem>
   371→                        ))}
   372→                      </SelectContent>
   373→                    </Select>
   374→                  </div>
   375→                  {data.languages.length > 0 && (
   376→                    <div className="flex flex-wrap gap-2">
   377→                      {data.languages.map((lang, i) => (
   378→                        <Badge key={i} variant="secondary" className="px-3 py-1.5 gap-1.5">
   379→                          {lang}
   380→                          <span className="text-xs text-muted-foreground">
   381→                            {data.languagesJa[i]}
   382→                          </span>
   383→                          <button
   384→                            onClick={() => removeLanguage(i)}
   385→                            className="ml-1 hover:text-destructive"
   386→                            aria-label="Remove"
   387→                          >
   388→                            <X className="h-3 w-3" />
   389→                          </button>
   390→                        </Badge>
   391→                      ))}
   392→                    </div>
   393→                  )}
   394→                </div>
   395→              </Field>
   396→            </div>
   397→          </Section>
   398→
   399→          {/* Education */}
   400→          <Section icon={GraduationCap} title="Education" desc="Academic history" action={<AddButton onClick={addEducation} />}>
   401→            <div className="space-y-4">
   402→              {data.education.length === 0 && <EmptyHint text="No education entries yet. Click Add to start." />}
   403→              {data.education.map((edu, i) => (
   404→                <Card key={i} onRemove={() => removeEducation(i)}>
   405→                  <div className="grid sm:grid-cols-2 gap-3">
   406→                    <Field label="Year / Month (graduation)">
   407→                      <Input value={edu.year} onChange={(e) => updateEducation(i, { year: e.target.value })} placeholder="2026" />
   408→                    </Field>
   409→                    <Field label="Degree (English)">
   410→                      <Input value={edu.degree} onChange={(e) => updateEducation(i, { degree: e.target.value })} placeholder="B.Tech" />
   411→                    </Field>
   412→                    <Field label="Degree (Japanese)">
   413→                      <Input value={edu.degreeJa ?? ""} onChange={(e) => updateEducation(i, { degreeJa: e.target.value })} placeholder="理学士" />
   414→                    </Field>
   415→                    <Field label="Field (English)">
   416→                      <Input value={edu.field} onChange={(e) => updateEducation(i, { field: e.target.value })} placeholder="Computer Science" />
   417→                    </Field>
   418→                    <Field label="Field (Japanese)">
   419→                      <Input value={edu.fieldJa ?? ""} onChange={(e) => updateEducation(i, { fieldJa: e.target.value })} placeholder="コンピュータサイエンス" />
   420→                    </Field>
   421→                    <Field label="Institution (English)">
   422→                      <Input value={edu.institution} onChange={(e) => updateEducation(i, { institution: e.target.value })} placeholder="SRM University" />
   423→                    </Field>
   424→                    <Field label="Institution (Japanese)">
   425→                      <Input value={edu.institutionJa ?? ""} onChange={(e) => updateEducation(i, { institutionJa: e.target.value })} placeholder="SRM大学" />
   426→                    </Field>
   427→                  </div>
   428→                </Card>
   429→              ))}
   430→            </div>
   431→          </Section>
   432→
   433→          {/* Projects */}
   434→          <Section icon={Briefcase} title="Projects" desc="Your technical projects" action={<AddButton onClick={addProject} />}>
   435→            <div className="space-y-4">
   436→              {data.projects.length === 0 && <EmptyHint text="No projects yet. Click Add to start." />}
   437→              {data.projects.map((proj, i) => (
   438→                <Card key={i} onRemove={() => removeProject(i)}>
   439→                  <div className="grid sm:grid-cols-2 gap-3">
   440→                    <Field label="Period">
   441→                      <Input value={proj.period} onChange={(e) => updateProject(i, { period: e.target.value })} placeholder="2024 Sep – 2024 Dec" />
   442→                    </Field>
   443→                    <Field label="Project name (English)">
   444→                      <Input value={proj.name} onChange={(e) => updateProject(i, { name: e.target.value })} placeholder="Collaboration Learning Platform" />
   445→                    </Field>
   446→                    <Field label="Project name (Japanese)">
   447→                      <Input value={proj.nameJa ?? ""} onChange={(e) => updateProject(i, { nameJa: e.target.value })} placeholder="コラボ学習プラットフォーム" />
   448→                    </Field>
   449→                    <Field label="Tech stack">
   450→                      <Input value={proj.techStack ?? ""} onChange={(e) => updateProject(i, { techStack: e.target.value })} placeholder="React.js, Node.js, Firebase, Tailwind CSS" />
   451→                    </Field>
   452→                  </div>
   453→                  <Field label="Description (English)" className="mt-3">
   454→                    <Textarea rows={3} value={proj.description} onChange={(e) => updateProject(i, { description: e.target.value })} placeholder="Developed a real-time learning platform with chat, live sessions, and auth..." />
   455→                  </Field>
   456→                  <Field label="Description (Japanese)" className="mt-2">
   457→                    <Textarea rows={3} value={proj.descriptionJa ?? ""} onChange={(e) => updateProject(i, { descriptionJa: e.target.value })} placeholder="チャット機能やライブセッション、認証機能を実装したリアルタイム学習プラットフォーム..." />
   458→                  </Field>
   459→                </Card>
   460→              ))}
   461→            </div>
   462→          </Section>
   463→
   464→          {/* Activities */}
   465→          <Section icon={Heart} title="Activities / Clubs" desc="Extracurricular and volunteer work" action={<AddButton onClick={addActivity} />}>
   466→            <div className="space-y-4">
   467→              {data.activities.length === 0 && <EmptyHint text="No activities yet. Click Add to start." />}
   468→              {data.activities.map((act, i) => (
   469→                <Card key={i} onRemove={() => removeActivity(i)}>
   470→                  <div className="grid sm:grid-cols-2 gap-3">
   471→                    <Field label="Period">
   472→                      <Input value={act.period} onChange={(e) => updateActivity(i, { period: e.target.value })} placeholder="9 months" />
   473→                    </Field>
   474→                    <Field label="Duration (Japanese)">
   475→                      <Input value={act.duration ?? ""} onChange={(e) => updateActivity(i, { duration: e.target.value })} placeholder="9か月" />
   476→                    </Field>
   477→                    <Field label="Organization (English)">
   478→                      <Input value={act.organization} onChange={(e) => updateActivity(i, { organization: e.target.value })} placeholder="Wellness Club" />
   479→                    </Field>
   480→                    <Field label="Organization (Japanese)">
   481→                      <Input value={act.organizationJa ?? ""} onChange={(e) => updateActivity(i, { organizationJa: e.target.value })} placeholder="ウェルネスクラブ" />
   482→                    </Field>
   483→                    <Field label="Role (English)">
   484→                      <Input value={act.role} onChange={(e) => updateActivity(i, { role: e.target.value })} placeholder="Member" />
   485→                    </Field>
   486→                    <Field label="Role (Japanese)">
   487→                      <Input value={act.roleJa ?? ""} onChange={(e) => updateActivity(i, { roleJa: e.target.value })} placeholder="メンバー" />
   488→                    </Field>
   489→                  </div>
   490→                  <Field label="Duties (English)" className="mt-3">
   491→                    <Textarea rows={2} value={act.duties} onChange={(e) => updateActivity(i, { duties: e.target.value })} placeholder="Organized stress-relief activities and fitness workshops..." />
   492→                  </Field>
   493→                  <Field label="Duties (Japanese)" className="mt-2">
   494→                    <Textarea rows={2} value={act.dutiesJa ?? ""} onChange={(e) => updateActivity(i, { dutiesJa: e.target.value })} placeholder="ストレス解消活動を企画・推進..." />
   495→                  </Field>
   496→                </Card>
   497→              ))}
   498→            </div>
   499→          </Section>
   500→
   501→          {/* Awards */}
   502→          <Section icon={Award} title="Awards / Achievements" desc="Honors and accomplishments" action={<AddButton onClick={addAward} />}>
   503→            <div className="space-y-4">
   504→              {data.awards.length === 0 && <EmptyHint text="No awards yet. Click Add to start." />}
   505→              {data.awards.map((aw, i) => (
   506→                <Card key={i} onRemove={() => removeAward(i)}>
   507→                  <div className="grid sm:grid-cols-2 gap-3">
   508→                    <Field label="Year">
   509→                      <Input value={aw.year} onChange={(e) => updateAward(i, { year: e.target.value })} placeholder="2025" />
   510→                    </Field>
   511→                    <Field label="Title (English)">
   512→                      <Input value={aw.title} onChange={(e) => updateAward(i, { title: e.target.value })} placeholder="Research Day Participant" />
   513→                    </Field>
   514→                    <Field label="Title (Japanese)">
   515→                      <Input value={aw.titleJa ?? ""} onChange={(e) => updateAward(i, { titleJa: e.target.value })} placeholder="第9回リサーチデイ参加" />
   516→                    </Field>
   517→                    <Field label="Organization (English)">
   518→                      <Input value={aw.organization} onChange={(e) => updateAward(i, { organization: e.target.value })} placeholder="SRM University" />
   519→                    </Field>
   520→                    <Field label="Organization (Japanese)">
   521→                      <Input value={aw.organizationJa ?? ""} onChange={(e) => updateAward(i, { organizationJa: e.target.value })} placeholder="SRM大学" />
   522→                    </Field>
   523→                  </div>
   524→                  <Field label="Description (English)" className="mt-3">
   525→                    <Textarea rows={2} value={aw.description} onChange={(e) => updateAward(i, { description: e.target.value })} placeholder="Participated in the 9th Research Day at SRM University-AP..." />
   526→                  </Field>
   527→                  <Field label="Description (Japanese)" className="mt-2">
   528→                    <Textarea rows={2} value={aw.descriptionJa ?? ""} onChange={(e) => updateAward(i, { descriptionJa: e.target.value })} placeholder="第9回リサーチデイに参加し、学術研究への関心を示した..." />
   529→                  </Field>
   530→                </Card>
   531→              ))}
   532→            </div>
   533→          </Section>
   534→
   535→          {/* Self-PR */}
   536→          <Section icon={Sparkles} title="Self-PR & Hobbies" desc="Your personal statement and interests">
   537→            <Field label="Self-PR (English)">
   538→              <Textarea rows={5} value={data.selfPr ?? ""} onChange={(e) => update("selfPr", e.target.value)} placeholder="I am a Computer Science student passionate about full-stack development and AI..." />
   539→            </Field>
   540→            <Field label="自己PR (Japanese)" className="mt-3">
   541→              <Textarea rows={5} value={data.selfPrJa ?? ""} onChange={(e) => update("selfPrJa", e.target.value)} placeholder="コンピュータサイエンスを専攻する学生です。フルスタック開発やAIプロジェクトに取り組む中で..." />
   542→            </Field>
   543→            <Field label="Hobbies (English)" className="mt-3">
   544→              <Input value={data.hobbies ?? ""} onChange={(e) => update("hobbies", e.target.value)} placeholder="Badminton, Fitness, Reading" />
   545→            </Field>
   546→            <Field label="趣味 (Japanese)" className="mt-2">
   547→              <Input value={data.hobbiesJa ?? ""} onChange={(e) => update("hobbiesJa", e.target.value)} placeholder="バドミントン、フィットネス、読書" />
   548→            </Field>
   549→          </Section>
   550→
   551→          {/* Save bar */}
   552→          <div className="sticky bottom-4 flex justify-end gap-2 print:hidden">
   553→            <Button onClick={save} disabled={saving} className="bg-brand-gradient text-white hover:opacity-90 font-semibold shadow-glow-brand">
   554→              {saving ? <Loader2 className="mr-2 h-4 w-4 animate-spin" /> : <Save className="mr-2 h-4 w-4" />}
   555→              Save Resume
   556→            </Button>
   557→          </div>
   558→        </motion.div>
   559→      )}
   560→
   561→      {/* Previews */}
   562→      {tab === "preview-ja" && <ResumePreview data={data} lang="ja" />}
   563→      {tab === "preview-en" && <ResumePreview data={data} lang="en" />}
   564→    </div>
   565→  );
   566→}
