   170→function computeCompletion(c: CandidateProfileDTO | null): number {
   171→  if (!c) return 0;
   172→  const checks: boolean[] = [
   173→    !!c.fullName?.trim(),
   174→    !!c.phone?.trim(),
   175→    !!c.bio?.trim(),
   176→    !!c.location?.trim(),
   177→    c.jlptLevel !== "NONE",
   178→    (c.skills?.length ?? 0) >= 3,
   179→    !!c.resumeUrl,
   180→    c.experienceYears > 0,
   181→    !!c.education && c.education.length > 0,
   182→    !!c.linkedinUrl?.trim(),
   183→  ];
   184→  const filled = checks.filter(Boolean).length;
   185→  return Math.round((filled / checks.length) * 100);
   186→}

   402→
   403→  const load = useCallback(async () => {
   404→    setLoading(true);
   405→    try {
   406→      const res = await api<{ applications: ApplicationDTO[] }>(
   407→        "/api/applications",
   408→      );
   409→      setApps(res.applications);
   410→    } catch {
   411→      setApps([]);
   412→    } finally {
   413→      setLoading(false);

   417→  useEffect(() => {
   418→    load();
   419→  }, [load]);

   421→  async function withdraw(a: ApplicationDTO) {
   422→    setBusyId(a.id);

   424→      await api(`/api/applications/${a.id}/status`, {
   425→        method: "PATCH",
   426→        body: JSON.stringify({ status: "WITHDRAWN" }),

   433→    } finally {
   434→      setBusyId(null);
   435→    }
   436→  }

   439→    return (
   440→      <div className="space-y-4">
   441→        <CardSkeleton lines={6} />

   450→        title={t("dash.apps.empty")}

   483→                <Fragment key={a.id}>
   484→                <TableRow>
   485→                  <TableCell className="pl-5 sm:pl-6">

   568→          </Table>
   569→        </div>
   570→      </SectionCard>
   571→    </div>
   572→  );

   577→function Profile() {
   578→  const candidate = useApp((s) => s.candidate);
   579→  const refreshAuth = useApp((s) => s.refreshAuth);
   580→  const { t } = useT();
   581→
   582→  const [form, setForm] = useState({
   583→    fullName: candidate?.fullName ?? "",
   584→    phone: candidate?.phone ?? "",
   585→    location: candidate?.location ?? "",
   586→    bio: candidate?.bio ?? "",
   587→    linkedinUrl: candidate?.linkedinUrl ?? "",
   588→    experienceYears: candidate?.experienceYears ?? 0,
   589→    jlptLevel: (candidate?.jlptLevel ?? "NONE") as JLPTLevel,
   590→    skills: candidate?.skills ?? [],
   591→    education: (candidate?.education ?? []) as EducationEntry[],
   592→  });
   593→  const [saving, setSaving] = useState(false);
   594→
   595→  // re-sync when store candidate changes (e.g. after refreshAuth)
   596→  useEffect(() => {
   597→    if (!candidate) return;
   598→    setForm((prev) => ({
   599→      ...prev,
   600→      fullName: candidate.fullName,
   601→      phone: candidate.phone ?? "",
   602→      location: candidate.location ?? "",
   603→      bio: candidate.bio ?? "",
   604→      linkedinUrl: candidate.linkedinUrl ?? "",
   605→      experienceYears: candidate.experienceYears,
   606→      jlptLevel: candidate.jlptLevel,
   607→      skills: candidate.skills,
   608→      education: candidate.education ?? [],
   609→    }));
   610→  }, [candidate?.updatedAt]);
   611→
   612→  function set<K extends keyof typeof form>(k: K, v: (typeof form)[K]) {
   613→    setForm((f) => ({ ...f, [k]: v }));
   614→  }
   615→
   616→  function addEducation() {
   617→    set("education", [
   618→      ...form.education,
   619→      { degree: "", field: "", institution: "", year: "" },
   620→    ]);
   621→  }
   622→  function removeEducation(idx: number) {
   623→    set(
   624→      "education",
   625→      form.education.filter((_, i) => i !== idx),
   626→    );
   627→  }
   628→  function setEducation(idx: number, patch: Partial<EducationEntry>) {
   629→    set(
   630→      "education",

   871→function Resume() {

   874→  const { t, locale } = useT();

   878→  async function uploadFile(file: File) {
   879→    if (file.type !== "application/pdf" && !file.name.endsWith(".pdf")) {
   880→      toast.error("Resume must be a PDF.");
   881→      return;
   882→    }
   883→    if (file.size > 5 * 1024 * 1024) {
   884→      toast.error("Max file size is 5MB.");
   885→      return;
   886→    }
   887→    setBusy(true);
   888→    setProgress(0);
   889→    try {
   890→      const fd = new FormData();
   891→      fd.append("file", file);
   892→      fd.append("kind", "resume");
   893→      // simulate upload progress (fetch doesn't give us easy progress)
   894→      const fakeTimer = setInterval(() => {
   895→        setProgress((p) => Math.min(90, (p ?? 0) + 10));
   896→      }, 80);
   897→      const res = await fetch("/api/upload", { method: "POST", body: fd });
   898→      clearInterval(fakeTimer);
   899→      setProgress(100);
   900→      if (!res.ok) {
   901→        const data = await res.json().catch(() => null);
   902→        throw new Error(data?.error || "Upload failed.");
   903→      }
   904→      toast.success("Resume uploaded.");
   905→      await refreshAuth();
   906→    } catch (e) {
   907→      toast.error(e instanceof Error ? e.message : "Upload failed.");
   908→    } finally {
   909→      setBusy(false);
   910→      setTimeout(() => setProgress(null), 600);
   911→    }
   912→  }
   913→
   914→  async function removeResume() {
   915→    setBusy(true);
   916→    try {
   917→      // The candidate PUT schema accepts `resumeUrl: null`. (The upload route
   918→      // sets resumeName separately — we only null the URL here, which is what
   919→      // applicants see.)
   920→      await api("/api/candidates/me", {
   921→        method: "PUT",
   922→        body: JSON.stringify({ resumeUrl: null }),
   923→      });
   924→      toast.success("Resume removed.");
   925→      await refreshAuth();
   926→    } catch (e) {
   927→      toast.error(e instanceof Error ? e.message : "Failed to remove.");
   928→    } finally {
   929→      setBusy(false);
   930→    }
   931→  }
   932→
   933→  return (
   934→    <div className="space-y-6 max-w-3xl">

   946→                Uploaded {formatRelative(candidate.updatedAt, locale)}

   950→              <Button asChild variant="outline" size="sm">
   951→                <a

   962→                  <Button

   966→                    className="text-muted-foreground hover:text-destructive"
