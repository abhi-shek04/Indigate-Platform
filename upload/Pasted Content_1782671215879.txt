import { NextRequest } from 'next/server'
import { getSession } from '@/lib/auth'
import { ok, err, handleError } from '@/lib/api'
import { db } from '@/lib/db'
import { supabaseAdmin } from '@/lib/supabase'

export async function POST(req: NextRequest) {
  try {
    const session = await getSession()
    if (!session) return err('Unauthorized.', 401)

    const formData = await req.formData()
    const file = formData.get('file') as File | null
    const kind = formData.get('kind') as string | null

    if (!file || !kind) return err('Missing file or kind.', 400)

    // Validate by kind
    const validations: Record<string, { roles: string[]; types: string[]; maxMB: number }> = {
      resume: { roles: ['CANDIDATE'], types: ['application/pdf'], maxMB: 5 },
      logo:   { roles: ['COMPANY'],   types: ['image/jpeg','image/png','image/webp'], maxMB: 2 },
      photo:  { roles: ['CANDIDATE'], types: ['image/jpeg','image/png','image/webp'], maxMB: 2 },
    }
    const rule = validations[kind]
    if (!rule) return err('Invalid upload kind.', 400)
    if (!rule.roles.includes(session.role)) return err('Forbidden.', 403)
    if (!rule.types.includes(file.type)) return err(`Invalid file type for ${kind}.`, 400)
    if (file.size > rule.maxMB * 1024 * 1024) return err(`File too large. Max ${rule.maxMB}MB.`, 400)

    // Build storage path
    const ext = file.name.split('.').pop() ?? 'bin'
    const path = `${kind}s/${session.id}/${Date.now()}.${ext}`

    // Convert File to Buffer
    const buffer = Buffer.from(await file.arrayBuffer())

    // Upload to Supabase Storage
    const bucket = process.env.SUPABASE_STORAGE_BUCKET!
    const { error: uploadError } = await supabaseAdmin.storage
      .from(bucket)
      .upload(path, buffer, { contentType: file.type, upsert: true })
    if (uploadError) throw new Error(uploadError.message)

    // Get public URL
    const { data } = supabaseAdmin.storage.from(bucket).getPublicUrl(path)
    const url = data.publicUrl

    // Save URL to DB based on kind
    if (kind === 'resume') {
      await db.candidateProfile.update({
        where: { userId: session.id },
        data: { resumeUrl: url, resumeName: file.name },
      })
    } else if (kind === 'logo') {
      await db.companyProfile.update({
        where: { userId: session.id },
        data: { logoUrl: url },
      })
    } else if (kind === 'photo') {
      await db.candidateProfile.update({
        where: { userId: session.id },
        data: { photoUrl: url },
      })
    }

    return ok({ url, name: file.name })
  } catch (e) {
    return handleError(e)
  }
}

const store = new Map<string, { count: number; resetAt: number }>()

export function rateLimit(key: string, max: number, windowMs: number): boolean {
  const now = Date.now()
  const entry = store.get(key)
  if (!entry || now > entry.resetAt) {
    store.set(key, { count: 1, resetAt: now + windowMs })
    return true
  }
  if (entry.count >= max) return false
  entry.count++
  return true
}

// Cleanup every 5 min
setInterval(() => {

  for (const [k, v] of store) if (now > v.resetAt) store.delete(k)

export async function POST(req: NextRequest, { params }: { params: Promise<{ id: string }> }) {
  try {
    const { id } = await params
    const cookieStore = await cookies()
    
    // Get or create anonymous session ID
    let sessionId = cookieStore.get('ig_anon')?.value
    if (!sessionId) {
      sessionId = crypto.randomUUID()
    }
    
    // Deduplicate: don't count same session twice in 24 hours
    const existing = await db.jobView.findFirst({
      where: { jobId: id, sessionId, viewedAt: { gte: new Date(Date.now() - 86400000) } }
    })
    
    const headers = new Headers()
    if (!cookieStore.get('ig_anon')?.value) {
      headers.set('Set-Cookie', `ig_anon=${sessionId}; Path=/; Max-Age=31536000; SameSite=Lax; HttpOnly`)
    }
    
    if (!existing) {
      await db.jobView.create({ data: { jobId: id, sessionId } })
    }
    
    return new Response(JSON.stringify({ counted: !existing }), { headers, status: 200 })
  } catch (e) {
