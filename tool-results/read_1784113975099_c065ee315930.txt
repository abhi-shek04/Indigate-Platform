    57→function StatCard({
    58→  value,
    59→  suffix,
    60→  label,
    61→  delay,
    62→  icon: Icon,
    63→}: {
    64→  value: number;
    65→  suffix?: string;
    66→  label: string;
    67→  delay: number;
    68→  icon: React.ComponentType<{ className?: string }>;
    69→}) {
    70→  const count = useCountUp(value, 1800);
    71→  const ref = useRef<HTMLDivElement>(null);
    72→  const [visible, setVisible] = useState(false);
    73→
    74→  useEffect(() => {
    75→    const el = ref.current;
    76→    if (!el) return;
    77→    const obs = new IntersectionObserver(
    78→      ([e]) => {
    79→        if (e.isIntersecting) {
    80→          setVisible(true);
    81→          obs.disconnect();
    82→        }
    83→      },
    84→      { threshold: 0.3 },
    85→    );
    86→    obs.observe(el);
    87→    return () => obs.disconnect();
    88→  }, []);
    89→
    90→  return (
    91→    <motion.div
    92→      ref={ref}
    93→      initial={{ opacity: 0, y: 24 }}
    94→      animate={visible ? { opacity: 1, y: 0 } : {}}
    95→      transition={{ duration: 0.6, ease: easeOutExpo, delay }}
    96→      whileHover={{ y: -4 }}
    97→      className="hero-stat group px-5 py-6 text-center transition-transform hover:-translate-y-1"
    98→    >
    99→      {/* Saffron accent line on top is provided by .hero-stat::after */}
   100→      <div className="relative flex flex-col items-center">
   101→        <motion.div
   102→          whileHover={{ rotate: [0, -8, 8, 0], scale: 1.08 }}
   103→          transition={{ duration: 0.5 }}
   104→          className="mb-3 grid place-items-center h-11 w-11 rounded-xl bg-saffron/10 text-saffron ring-1 ring-inset ring-saffron/20"
   105→        >
   106→          <Icon className="h-5 w-5" />
   107→        </motion.div>
   108→        <div className="font-display text-4xl sm:text-5xl font-extrabold tracking-tight leading-none">
   109→          <span className="text-gradient-brand">
   110→            {count.toLocaleString()}
   111→            {suffix}
   112→          </span>
   113→        </div>
   114→        <p className="mt-2 text-xs sm:text-sm font-medium text-muted-foreground uppercase tracking-wide">
   115→          {label}
   116→        </p>
   117→      </div>
   118→    </motion.div>
   119→  );
   120→}
